“System.Text.Json Polymorphic Deserialization Strategies”

When serializing or deserializing JSON that represents objects from a type hierarchy, you often need a mechanism to determine the specific derived type based on the JSON content. System.Text.Json offers several approaches to handle this, ranging from manual converter implementations to built-in attributes introduced in later .NET versions.

Solutions for Polymorphic Deserialization in System.Text.Json

Here are various strategies discussed by developers to achieve polymorphic deserialization:

Solution 1: Custom JsonConverter with Type Discriminator

This approach involves creating a custom JsonConverter that inspects a specific property in the JSON payload (a “type discriminator”) to decide which derived type to deserialize into. This is a flexible method, especially for older .NET versions.

Example Classes:

public class BaseClass
{
    public int Int { get; set; }
}

public class DerivedA : BaseClass
{
    public string Str { get; set; }
}

public class DerivedB : BaseClass
{
    public bool Bool { get; set; }
}

Custom Converter (BaseClassConverter):

This converter reads a TypeDiscriminator property to identify the concrete type.

public class BaseClassConverter : JsonConverter<BaseClass>
{
    private enum TypeDiscriminator
    {
        BaseClass = 0,
        DerivedA = 1,
        DerivedB = 2
    }

    public override bool CanConvert(Type type)
    {
        return typeof(BaseClass).IsAssignableFrom(type);
    }

    public override BaseClass Read(
        ref Utf8JsonReader reader,
        Type typeToConvert,
        JsonSerializerOptions options)
    {
        if (reader.TokenType != JsonTokenType.StartObject)
        {
            throw new JsonException();
        }

        // Advance to find the "TypeDiscriminator" property
        while (reader.Read())
        {
            if (reader.TokenType == JsonTokenType.PropertyName && reader.GetString() == "TypeDiscriminator")
            {
                if (!reader.Read() || reader.TokenType != JsonTokenType.Number)
                {
                    throw new JsonException("Invalid TypeDiscriminator value.");
                }

                BaseClass baseClass;
                TypeDiscriminator typeDiscriminator = (TypeDiscriminator)reader.GetInt32();

                // Skip to the start of the actual object for the derived type
                if (!reader.Read() || reader.TokenType != JsonTokenType.PropertyName || reader.GetString() != "TypeValue")
                {
                    throw new JsonException("Expected 'TypeValue' property.");
                }
                if (!reader.Read() || reader.TokenType != JsonTokenType.StartObject)
                {
                    throw new JsonException("Expected start of nested object for TypeValue.");
                }

                switch (typeDiscriminator)
                {
                    case TypeDiscriminator.DerivedA:
                        baseClass = JsonSerializer.Deserialize<DerivedA>(ref reader, options);
                        break;
                    case TypeDiscriminator.DerivedB:
                        baseClass = JsonSerializer.Deserialize<DerivedB>(ref reader, options);
                        break;
                    default:
                        throw new NotSupportedException();
                }

                // Ensure we are at the end of the current object before returning
                if (!reader.Read() || reader.TokenType != JsonTokenType.EndObject)
                {
                    throw new JsonException("Expected end of object.");
                }
                return baseClass;
            }
            reader.Skip(); // Skip other properties until TypeDiscriminator
        }
        throw new JsonException("Could not find TypeDiscriminator property.");
    }

    public override void Write(
        Utf8JsonWriter writer,
        BaseClass value,
        JsonSerializerOptions options)
    {
        writer.WriteStartObject();

        if (value is DerivedA derivedA)
        {
            writer.WriteNumber("TypeDiscriminator", (int)TypeDiscriminator.DerivedA);
            writer.WritePropertyName("TypeValue");
            JsonSerializer.Serialize(writer, derivedA, options);
        }
        else if (value is DerivedB derivedB)
        {
            writer.WriteNumber("TypeDiscriminator", (int)TypeDiscriminator.DerivedB);
            writer.WritePropertyName("TypeValue");
            JsonSerializer.Serialize(writer, derivedB, options);
        }
        else
        {
            throw new NotSupportedException();
        }

        writer.WriteEndObject();
    }
}

Usage Example:

var options = new JsonSerializerOptions
{
    Converters = { new BaseClassConverter() },
    WriteIndented = true
};

var objects = new List<BaseClass> { new DerivedA { Int = 1, Str = "Sample A" }, new DerivedB { Int = 2, Bool = true } };
string jsonString = JsonSerializer.Serialize(objects, options);
Console.WriteLine(jsonString);

var roundTrip = JsonSerializer.Deserialize<List<BaseClass>>(jsonString, options);

Considerations: This manual approach is verbose and requires careful handling of the JSON reader. For more robust solutions, explore attributes or newer .NET features.

Solution 2: Using JsonDerivedTypeAttribute (.NET 7+)

.NET 7 introduced built-in support for polymorphic serialization and deserialization using attributes. You can decorate your base class with [JsonDerivedType] to register known derived types.

Example Classes:

[JsonDerivedType(typeof(DerivedType1), "Type1")] // Discriminator value can be string or integer
[JsonDerivedType(typeof(DerivedType2), "Type2")]
public abstract class BaseType { }

public class DerivedType1 : BaseType { public string Derived1 { get; set; } }
public class DerivedType2 : BaseType { public int Derived2 { get; set; } }

Serialization:

var list = new List<BaseType> { new DerivedType1 { Derived1 = "value 1" }, new DerivedType2 { Derived2 = 123 } };
var json = JsonSerializer.Serialize(list);
Console.WriteLine(json);
// Output (example with "$type" discriminator):
// [
//  {
//    "$type": "Type1",
//    "Derived1": "value 1"
//  },
//  {
//    "$type": "Type2",
//    "Derived2": 123
//  }
// ]

Deserialization:

The Deserialize method will automatically use the registered types.

var deserializedList = JsonSerializer.Deserialize<List<BaseType>>(json);

Runtime Configuration:

If you cannot modify the base class, you can configure polymorphism at runtime using DefaultJsonTypeInfoResolver and JsonTypeInfo.PolymorphismOptions.

var resolver = new DefaultJsonTypeInfoResolver
{
    Modifiers =
    {
        static typeInfo =>
        {
            if (typeInfo.Type == typeof(BaseType))
            {
                typeInfo.PolymorphismOptions = new()
                {
                    // Register derived types and their discriminator values
                    DerivedTypes =
                    {
                        new JsonDerivedType(typeof(DerivedType1), "Type1"),
                        new JsonDerivedType(typeof(DerivedType2), "Type2")
                    }
                };
            }
        }
    }
};

var options = new JsonSerializerOptions { TypeInfoResolver = resolver };
var json = JsonSerializer.Serialize(list, options);
var deserializedList = JsonSerializer.Deserialize<List<BaseType>>(json, options);

Security: The attribute-based approach is safer as it only serializes/deserializes whitelisted types, preventing potential type injection vulnerabilities found in some older serialization mechanisms like Newtonsoft.Json’s TypeNameHandling.All. For details on security concerns, refer to GitHub Issue #41347 .

Solution 3: Generic TypeDiscriminatorConverter

This solution provides a reusable converter that discovers derived types via reflection and uses a dedicated ITypeDiscriminator interface to identify the type.

Interface and Converter:

public interface ITypeDiscriminator
{
    string TypeDiscriminator { get; }
}

public class TypeDiscriminatorConverter<T> : JsonConverter<T> where T : ITypeDiscriminator
{
    private readonly IEnumerable<Type> _types;

    public TypeDiscriminatorConverter()
    {
        var baseType = typeof(T);
        // Discover derived types - caution with large projects due to reflection overhead
        _types = AppDomain.CurrentDomain.GetAssemblies()
            .SelectMany(s => s.GetTypes())
            .Where(p => baseType.IsAssignableFrom(p) && p.IsClass && !p.IsAbstract)
            .ToList();
    }

    public override T Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
    {
        if (reader.TokenType != JsonTokenType.StartObject) throw new JsonException();

        using var jsonDocument = JsonDocument.ParseValue(ref reader);
        if (!jsonDocument.RootElement.TryGetProperty(nameof(ITypeDiscriminator.TypeDiscriminator), out var typeProperty))
        {
            throw new JsonException();
        }

        var targetType = _types.FirstOrDefault(x => x.Name == typeProperty.GetString());
        if (targetType == null) throw new JsonException();

        var jsonObject = jsonDocument.RootElement.GetRawText();
        return (T)JsonSerializer.Deserialize(jsonObject, targetType, options);
    }

    public override void Write(Utf8JsonWriter writer, T value, JsonSerializerOptions options)
    {
        JsonSerializer.Serialize(writer, (object)value, options);
    }
}

Example Usage:

public interface ISurveyStepResult : ITypeDiscriminator { string Id { get; set; } }
public class BoolStepResult : ISurveyStepResult { public string Id { get; set; } public string TypeDiscriminator => nameof(BoolStepResult); public bool Value { get; set; } }
public class TextStepResult : ISurveyStepResult { public string Id { get; set; } public string TypeDiscriminator => nameof(TextStepResult); public string Value { get; set; } }

var options = new JsonSerializerOptions { Converters = { new TypeDiscriminatorConverter<ISurveyStepResult>() }, WriteIndented = true };
var surveyResult = new { Steps = new List<ISurveyStepResult> { new BoolStepResult { Id = "1", Value = true }, new TextStepResult { Id = "2", Value = "text" } } };
var json = JsonSerializer.Serialize(surveyResult, options);
var back = JsonSerializer.Deserialize(json, options);

Notes: Discovering types via reflection can have performance implications. Consider caching or alternative registration methods for performance-critical applications.

Solution 4: Custom JsonConverterFactory for Reusability

This approach builds upon custom converters by creating a JsonConverterFactory for better organization and reuse, especially when dealing with multiple abstract types.

Attributes and Factory:

[AttributeUsage(AttributeTargets.Class, AllowMultiple = false)]
public class DiscriminatorAttribute : Attribute { public string Property { get; } public DiscriminatorAttribute(string property) => Property = property; }

[AttributeUsage(AttributeTargets.Class, AllowMultiple = false)]
public class DiscriminatorValueAttribute : Attribute { public object Value { get; } public DiscriminatorValueAttribute(object value) => Value = value; }

// Abstract Base Class Example
[Discriminator(nameof(Type))]
public abstract class Identity { public virtual IdentityType Type { get; protected set; } }
public enum IdentityType { Person, Organization }

[DiscriminatorValue(IdentityType.Person)]
public class Person : Identity { public string Name { get; set; } }

// Converter Factory and Converter
public class AbstractClassConverterFactory : JsonConverterFactory
{
    private static readonly Dictionary<Type, JsonConverter> Converters = new Dictionary<Type, JsonConverter>();
    private readonly JsonNamingPolicy _namingPolicy;

    public AbstractClassConverterFactory(JsonNamingPolicy namingPolicy) => _namingPolicy = namingPolicy;

    public override bool CanConvert(Type typeToConvert) => typeToConvert.IsClass && typeToConvert.IsAbstract && typeToConvert.IsDefined(typeof(DiscriminatorAttribute));

    public override JsonConverter CreateConverter(Type typeToConvert, JsonSerializerOptions options)
    {
        if (!Converters.TryGetValue(typeToConvert, out var converter))
        {
            var converterType = typeof(AbstractClassConverter<>).MakeGenericType(typeToConvert);
            converter = (JsonConverter)Activator.CreateInstance(converterType, _namingPolicy);
            Converters.Add(typeToConvert, converter);
        }
        return converter;
    }
}

public class AbstractClassConverter<T> : JsonConverter<T>
{
    private readonly PropertyInfo _discriminatorProperty;
    private readonly Dictionary<string, Type> _typeMappings;
    private readonly JsonNamingPolicy _namingPolicy;

    public AbstractClassConverter(JsonNamingPolicy namingPolicy)
    {
        _namingPolicy = namingPolicy;
        var discriminatorAttribute = typeof(T).GetCustomAttribute<DiscriminatorAttribute>() ?? throw new InvalidOperationException("DiscriminatorAttribute not found.");
        _discriminatorProperty = typeof(T).GetProperty(discriminatorAttribute.Property) ?? throw new InvalidOperationException("Discriminator property not found.");
        _typeMappings = FindDerivedTypeMappings();
    }

    private Dictionary<string, Type> FindDerivedTypeMappings()
    {
        var mappings = new Dictionary<string, Type>();
        foreach (var derivedType in typeof(T).Assembly.GetTypes().Where(t => t.IsClass && !t.IsAbstract && t.IsSubclassOf(typeof(T))))
        {
            var discriminatorValueAttribute = derivedType.GetCustomAttribute<DiscriminatorValueAttribute>();
            if (discriminatorValueAttribute != null)
            {
                var discriminatorValue = discriminatorValueAttribute.Value.ToString();
                mappings.Add(discriminatorValue, derivedType);
            }
        }
        return mappings;
    }

    public override T Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
    {
        if (reader.TokenType != JsonTokenType.StartObject) throw new JsonException();

        using var jsonDocument = JsonDocument.ParseValue(ref reader);
        var discriminatorPropertyName = _namingPolicy?.ConvertName(_discriminatorProperty.Name) ?? _discriminatorProperty.Name;

        if (!jsonDocument.RootElement.TryGetProperty(discriminatorPropertyName, out var discriminatorProperty))
            throw new JsonException($"Discriminator property '{discriminatorAttribute.Property}' not found.");

        var discriminatorValue = discriminatorProperty.GetString();
        if (!_typeMappings.TryGetValue(discriminatorValue, out var derivedType))
            throw new JsonException($"Unknown discriminator value: {discriminatorValue}");

        return (T)JsonSerializer.Deserialize(jsonDocument.RootElement.GetRawText(), derivedType, options);
    }

    public override void Write(Utf8JsonWriter writer, T value, JsonSerializerOptions options)
    {
        JsonSerializer.Serialize(writer, value, options); // Let the default serializer handle derived types
    }
}

Registration:

// In Startup.cs or Program.cs
services.AddControllersWithViews()
    .AddJsonOptions(options =>
    {
        options.JsonSerializerOptions.Converters.Add(new AbstractClassConverterFactory(options.JsonSerializerOptions.PropertyNamingPolicy));
    });

Solution 6: Using JsonDocument and Manual Deserialization

For scenarios where the type discriminator is already a property of the main object, you can parse the JSON as a JsonDocument, inspect the discriminator, and then deserialize the relevant parts.

Example with CreditReportConverter:

This converter uses a status property to determine the actual type.

public enum Status { Complete, Failed, Pending }

public abstract record Result; // Base type
public record Success(string Data) : Result;
public record Failure(string ErrorMessage) : Result;

public class CreditReportConverter : JsonConverter<Result>
{
    public override bool CanConvert(Type type) => type.IsAssignableTo(typeof(Result));

    public override Result Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
    {
        using var json = JsonDocument.ParseValue(ref reader);

        if (!json.RootElement.TryGetProperty("status", out var statusElement))
            throw new JsonException("Missing 'status' property.");

        if (!Enum.TryParse(statusElement.GetString(), out Status status))
            throw new JsonException($"Unrecognized status: {statusElement.GetString()}");

        return status switch
        {
            Status.Complete => json.Deserialize<Success>(options) ?? throw new JsonException("Failed to deserialize Success."),
            Status.Failed => json.Deserialize<Failure>(options) ?? throw new JsonException("Failed to deserialize Failure."),
            _ => json.Deserialize<Result>(options) ?? throw new JsonException("Failed to deserialize Result.")
        };
    }

    public override void Write(Utf8JsonWriter writer, Result report, JsonSerializerOptions options)
    {
        string json = report switch
        {
            Success s => JsonSerializer.Serialize(s, options),
            Failure f => JsonSerializer.Serialize(f, options),
            _ => JsonSerializer.Serialize(report, options)
        };
        writer.WriteRawValue(json);
    }
}

Usage:

var options = new JsonSerializerOptions { Converters = { new CreditReportConverter() }, WriteIndented = true };
// ... serialize/deserialize credit reports

Solution 7: Casting to object for Serialization

When serializing an instance of a derived class that’s treated as its base type, casting to object can sometimes reveal the derived type’s properties. This is a serialization-only trick and doesn’t solve deserialization challenges directly.

class BaseClass { public int Id { get; set; } }
class Product : BaseClass { public string Name { get; set; } public decimal Price { get; set; } }

BaseClass product = new Product { Id = 1, Name = "Apple", Price = 999.00m };
var json = JsonSerializer.Serialize((object)product, new JsonSerializerOptions { WriteIndented = true });
// Output includes Name and Price

Choosing the Right Approach

  • .NET 7+: JsonDerivedTypeAttribute is the most modern and recommended approach for new projects due to its built-in support, clarity, and security benefits.
  • Older .NET Versions or Complex Scenarios: Custom JsonConverters offer the most flexibility. Consider abstracting them using a JsonConverterFactory for better code organization.
  • Code Generation: For scenarios demanding minimal boilerplate and automated type discovery, source generators are an excellent option.
  • Simple Type Mapping: If you’re mapping a few specific interfaces to concrete implementations without complex hierarchies, a simple StaticTypeMapConverter might suffice.

When implementing custom converters, be mindful of the Utf8JsonReader’s forward-only nature. Strategies involving JsonDocument or careful reader manipulation are often necessary for scenarios where the type discriminator needs to be read before the main object’s properties.

Consider the security implications of polymorphic deserialization. Always prefer mechanisms that explicitly whitelist types to mitigate risks like type injection attacks.

For further exploration into System.Text.Json features, you can refer to the official Microsoft documentation and related articles on handling JSON serialization in .NET.

FAQs on System.Text.Json Polymorphic Deserialization

Q: What is polymorphic deserialization in System.Text.Json

ANS: Polymorphic deserialization is the process of deserializing JSON data into a hierarchy of types, where the specific derived type is determined at runtime based on information within the JSON payload, such as a type discriminator property.

Q: Is polymorphic deserialization supported in .NET 7 and later versions

ANS: Yes, .NET 7 introduced built-in support for polymorphic serialization and deserialization using attributes like JsonDerivedTypeAttribute, significantly simplifying the process.

Q: Can I implement polymorphic deserialization without .NET 7 features

ANS: Yes, you can implement polymorphic deserialization by creating custom JsonConverter classes that inspect a type discriminator property in the JSON to select the correct derived type for deserialization.

Q: What are the security implications of polymorphic deserialization

ANS: Allowing untrusted input to dictate the deserialized type can lead to security vulnerabilities like type injection. It’s crucial to use mechanisms that whitelist acceptable derived types, such as JsonDerivedTypeAttribute or custom converters that explicitly define allowed types.

Q: How do I specify a type discriminator property in System.Text.Json

ANS: In .NET 7+, you can use the JsonDerivedTypeAttribute on the base class, specifying a string or integer value for the TypeDiscriminator property. For custom converters, you define the property name within your converter logic.

Q: Can I use interfaces for polymorphic deserialization

ANS: Yes, you can implement polymorphic deserialization with interfaces by creating custom converters that map interface types to concrete implementations based on a discriminator. The principles are similar to handling abstract base classes.

Q: What happens if an unknown type is encountered during deserialization

ANS: The behavior depends on the implementation. Custom converters might throw exceptions, while .NET 7+ features allow configuration via JsonUnknownDerivedTypeHandling (e.g., FailSerialization, FallBackToBaseType).

Q: How can I avoid reflection for type discovery in custom converters

ANS: Instead of reflection, you can manually register derived types with your converter or converter factory, or use attributes like JsonDerivedTypeAttribute where the .NET runtime handles type mapping.
public class BaseClass
{
    public int Int { get; set; }
}
public class DerivedA : BaseClass
{
    public string Str { get; set; }
}
public class DerivedB : BaseClass
{
    public bool Bool { get; set; }
}
public class BaseClassConverter : JsonConverter&lt;BaseClass&gt;
{
    private enum TypeDiscriminator
    {
        BaseClass = 0,
        DerivedA = 1,
        DerivedB = 2
    }

    public override bool CanConvert(Type type)
    {
        return typeof(BaseClass).IsAssignableFrom(type);
    }
    
    public override BaseClass Read(
        ref Utf8JsonReader reader,
        Type typeToConvert,
        JsonSerializerOptions options)
    {
        if (reader.TokenType != JsonTokenType.StartObject)
        {
            throw new JsonException();
        }

        if (!reader.Read()
                || reader.TokenType != JsonTokenType.PropertyName
                || reader.GetString() != &quot;TypeDiscriminator&quot;)
        {
            throw new JsonException();
        }
        
        if (!reader.Read() || reader.TokenType != JsonTokenType.Number)
        {
            throw new JsonException();
        }

        BaseClass baseClass;
        TypeDiscriminator typeDiscriminator = (TypeDiscriminator)reader.GetInt32();
        switch (typeDiscriminator)
        {
            case TypeDiscriminator.DerivedA:
                if (!reader.Read() || reader.GetString() != &quot;TypeValue&quot;)
                {
                    throw new JsonException();
                }
                if (!reader.Read() || reader.TokenType != JsonTokenType.StartObject)
                {
                    throw new JsonException();
                }
                baseClass = (DerivedA)JsonSerializer.Deserialize(ref reader, typeof(DerivedA));
                break;
            case TypeDiscriminator.DerivedB:
                if (!reader.Read() || reader.GetString() != &quot;TypeValue&quot;)
                {
                    throw new JsonException();
                }
                if (!reader.Read() || reader.TokenType != JsonTokenType.StartObject)
                {
                    throw new JsonException();
                }
                baseClass = (DerivedB)JsonSerializer.Deserialize(ref reader, typeof(DerivedB));
                break;
            default:
                throw new NotSupportedException();
        }

        if (!reader.Read() || reader.TokenType != JsonTokenType.EndObject)
        {
            throw new JsonException();
        }

        return baseClass;
    }

    public override void Write(
        Utf8JsonWriter writer,
        BaseClass value,
        JsonSerializerOptions options)
    {
        writer.WriteStartObject();

        if (value is DerivedA derivedA)
        {
            writer.WriteNumber(&quot;TypeDiscriminator&quot;, (int)TypeDiscriminator.DerivedA);
            writer.WritePropertyName(&quot;TypeValue&quot;);
            JsonSerializer.Serialize(writer, derivedA);
        }
        else if (value is DerivedB derivedB)
        {
            writer.WriteNumber(&quot;TypeDiscriminator&quot;, (int)TypeDiscriminator.DerivedB);
            writer.WritePropertyName(&quot;TypeValue&quot;);
            JsonSerializer.Serialize(writer, derivedB);
        }
        else
        {
            throw new NotSupportedException();
        }

        writer.WriteEndObject();
    }
}
