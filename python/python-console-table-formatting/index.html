“Python: Create and Format Tables in Console Output”

Displaying data in a structured, readable table format in the console is a common requirement for many Python applications, whether it’s for configuration output, logging, or presenting query results. Python offers several methods and libraries to achieve this, ranging from simple string formatting to sophisticated table generators. This exploration covers various approaches to create and style tables in your terminal.

Key Python Libraries for Table Formatting

Several Python packages are designed to simplify the creation of console tables. Each offers unique features and styling options:

  • tabulate: Known for its simplicity and extensive table formatting options, including support for various styles like plain, simple, grid, pipe, orgtbl, rst, and more. It’s excellent for quickly generating visually appealing tables from lists of lists or dictionaries.

    • Find it on PyPI .
    from tabulate import tabulate

headers = ['Name', 'Age']

print(tabulate(data, headers=headers))

print("\nOrg Table Format:")

def format_matrix(header, matrix,
                  top_format, left_format, cell_format, row_delim, col_delim):
    table = [[''] + header] + [[name] + row for name, row in zip(header, matrix)]
    table_format = [['{:^{}}'] + len(header) * [top_format]] \
                 + len(matrix) * [[left_format] + len(header) * [cell_format]]
    col_widths = [max(
                      len(format.format(cell, 0))
                      for format, cell in zip(col_format, col))
                  for col_format, col in zip(zip(*table_format), zip(*table))]
    return row_delim.join(
               col_delim.join(
                   format.format(cell, width)
                   for format, cell, width in zip(row_format, row, col_widths))
               for row_format, row in zip(table_format, table))

def print_table(data, cols, wide):
    '''Prints formatted data on columns of given width.'''
    n, r = divmod(len(data), cols)
    pat = '{{:{}}}'.format(wide)
    line = '\n'.join(pat * cols for _ in range(n))
    last_line = pat * r
    print(line.format(*data))
    print(last_line.format(*data[n*cols:]))

def pretty_print_table(rows, line_between_rows=True):
  &quot;&quot;&quot;
  Example Output
  ┌──────┬─────────────┬────┬───────┐
  │ True │ short       │ 77 │ catty │
  ├──────┼─────────────┼────┼───────┤
  │ 36   │ long phrase │ 9  │ dog   │
  ├──────┼─────────────┼────┼───────┤
  │ 8    │ medium      │ 3  │ zebra │
  └──────┴─────────────┴────┴───────┘
  &quot;&quot;&quot;

  # find the max length of each column
  max_col_lens = list(map(max, zip(*[(len(str(cell)) for cell in row) for row in rows])))

  # print the table's top border
  print('┌' + '┬'.join('─' * (n + 2) for n in max_col_lens) + '┐')

  rows_separator = '├' + '┼'.join('─' * (n + 2) for n in max_col_lens) + '┤'

  row_fstring = ' │ '.join(&quot;{: &lt;%s}&quot; % n for n in max_col_lens)

  for i, row in enumerate(rows):
    print('│', row_fstring.format(*map(str, row)), '│')
    
    if line_between_rows and i &lt; len(rows) - 1:
      print(rows_separator)

  # print the table's bottom border
  print('└' + '┴'.join('─' * (n + 2) for n in max_col_lens) + '┘')

class AlignedPrinter:
    def __init__(self):
        self._col_widths = {}
        self._lines = []

    def __del__(self):
        self.aligned_print()

    def print(self, *args):
        line = ' '.join([str(x) for x in args])
        cols = line.split('\t')
        self._lines.append(cols)
        for i in range(len(cols)):
            self._col_widths[i] = max(len(cols[i]), self._col_widths.get(i, 0))

    def aligned_print(self):
        for line in self._lines:
            aligned = []
            for i in range(len(line)):
                col = line[i]
                aligned.append(col + ' ' * (self._col_widths[i] - len(col)))
            print(' '.join(aligned))
        self.__init__()

def printTable(table,align=&quot;&quot;,hasHeader=False,pad=2,isGrid=False):
    table = [row[:] for row in table] # copy table
    numRows,numCols = len(table),len(table[0]) # table size
    align = align.ljust(numCols,&quot;L&quot;) # align left by default
    align = [&quot;RC&quot;.find(c)+1 for c in align] # convert to index (?RC=012)
    widths = [max(len(row[col]) for row in table) for col in range(numCols)] # column widths

    # --- apply column widths with alignments ---
    if hasHeader: # header is centered
        for x in range(numCols): table[0][x] = table[0][x].center(widths[x])
    for y in range(hasHeader,numRows): # apply column alignments
        for x in range(numCols): c = table[y][x]; table[y][x] = [c.ljust,c.rjust,c.center][align[x]](widths[x])

    # --- data for printing
    P = &quot; &quot;*pad; LSEP,SEP,RSEP = &quot;│&quot;+P, P+&quot;│&quot;+P, P+&quot;│&quot;
    lines = [&quot;─&quot;*(widths[col]+pad*2) for col in range(numCols)]

    drawLine = [isGrid]*numRows; drawLine[0]|=hasHeader; drawLine[-1] = False
    if hasHeader or isGrid: gridLine = &quot;├&quot;+&quot;┼&quot;.join(lines)+&quot;┤&quot; # if any(drawLine)

    # --- print rows ---
    print(&quot;┌&quot;+&quot;┬&quot;.join(lines)+&quot;┐&quot;)
    for y in range(numRows):
        print(LSEP+SEP.join(table[y])+RSEP)
        if drawLine[y]: print(gridLine)
    print(&quot;└&quot;+&quot;┴&quot;.join(lines)+&quot;┘&quot;)
