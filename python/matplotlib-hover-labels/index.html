“Python Matplotlib: Displaying Labels on Hover for Plots”

When visualizing data with Matplotlib in Python, it’s often beneficial to provide interactive elements that reveal more information when a user hovers their mouse over specific data points. This can significantly enhance the data exploration experience, making it easier to identify individual data points and their associated values. Several methods exist to achieve this, ranging from custom event handling to leveraging specialized libraries.

Here are various approaches to implement hover functionality for your Matplotlib plots:

Solution 1: Custom Hover Annotation for Scatter and Line Plots

This solution involves creating custom functions to manage the display of annotations when the mouse hovers over plot elements. It directly manipulates Matplotlib’s event handling system.

For scatter plots:

import matplotlib.pyplot as plt
import numpy as np; np.random.seed(1)

# Sample data for scatter plot
x_scatter = np.random.rand(15)
y_scatter = np.random.rand(15)
names_scatter = np.array(list("ABCDEFGHIJKLMNO"))
colors_scatter = np.random.randint(1,5,size=15)

norm = plt.Normalize(1,4)
cmap = plt.cm.RdYlGn

fig_scatter, ax_scatter = plt.subplots()
scatter_plot = ax_scatter.scatter(x_scatter,y_scatter,c=colors_scatter, s=100, cmap=cmap, norm=norm)

# Initialize annotation object
annotation = ax_scatter.annotate("", xy=(0,0), xytext=(20,20),textcoords="offset points",
                    bbox=dict(boxstyle="round", fc="w"),
                    arrowprops=dict(arrowstyle="-&gt;"))
annotation.set_visible(False)

def update_annotation(ind_data):
    """Updates the annotation text and position."""
    pos = scatter_plot.get_offsets()[ind_data["ind"][0]]
    annotation.xy = pos
    text_content = "{}, {}".format(" ".join(list(map(str,ind_data["ind"]))),
                           " ".join([names_scatter[n] for n in ind_data["ind"]]))
    annotation.set_text(text_content)
    annotation.get_bbox_patch().set_facecolor(cmap(norm(colors_scatter[ind_data["ind"][0]])))
    annotation.get_bbox_patch().set_alpha(0.4)

def hover_scatter(event):
    """Handles mouse motion events for the scatter plot."""
    is_visible = annotation.get_visible()
    if event.inaxes == ax_scatter:
        is_over_plot, current_data_indices = scatter_plot.contains(event)
        if is_over_plot:
            update_annotation(current_data_indices)
            annotation.set_visible(True)
            fig_scatter.canvas.draw_idle()
        else:
            if is_visible:
                annotation.set_visible(False)
                fig_scatter.canvas.draw_idle()

fig_scatter.canvas.mpl_connect("motion_notify_event", hover_scatter)
plt.title("Scatter Plot with Hover Annotations")
plt.show()

For line plots:

import matplotlib.pyplot as plt

Solution 2: Hovering Over Individual Lines Without Clicking

This method allows for hovering over distinct lines within a plot without requiring a click event.

import matplotlib.pyplot as plt

fig, ax = plt.subplots()

# Create multiple plot lines with unique identifiers
for i in range(4):
    ax.plot(
        [i*1,i*2,i*3,i*4],
        gid=i # Assign a unique identifier
    )

def on_plot_hover(event):
    """Detects which plot line is under the mouse."""
    # Iterate through each plotted line
    for curve in ax.get_lines():
        # Check if the mouse cursor is over the line
        if curve.contains(event)[0]:
            print(f"Hovering over line with GID: {curve.get_gid()}")

# Connect the hover event to the handler function
fig.canvas.mpl_connect('motion_notify_event', on_plot_hover)
plt.title("Hover Over Individual Lines")
plt.show()

Solution 3: Using the mplcursors Package

The mplcursors library offers a streamlined approach to adding interactive data cursors, including tooltips, to Matplotlib plots. It’s often the most straightforward method.

First, install the package: Use conda install -c conda-forge mplcursors if you are using Anaconda, or pip install mplcursors otherwise.

Ensure your plots are rendered in an interactive window (e.g., by executing %matplotlib qt in a Jupyter Notebook cell).

import matplotlib.pyplot as plt
import pandas_datareader as web  # For sample data
from mplcursors import cursor

# Fetch sample data (e.g., stock prices)
try:
    df = web.DataReader('AAPL', data_source='yahoo', start='2021-03-09', end='2022-06-13')
except Exception as e:
    print(f"Could not fetch data: {e}. Using dummy data instead.")
    # Fallback to dummy data if data fetching fails
    dates = plt.range(5)
    df = type('obj', (object,), {'Close': [100, 105, 102, 108, 110], 'index': dates})()


plt.figure(figsize=(12, 7))
plt.plot(df.index, df.Close)
# Enable cursor with hover functionality
cursor(hover=True)
plt.title("Stock Price with mplcursors Tooltips")
plt.show()

# Example with Pandas plotting
ax_pandas = df.plot(y='Close', figsize=(10, 7))
cursor(hover=True)
plt.title("Pandas Plot with mplcursors Tooltips")
plt.show()

# Example with Seaborn
import seaborn as sns
tips_data = sns.load_dataset('tips')

sns.relplot(data=tips_data, x="total_bill", y="tip", hue="day", col="time")
cursor(hover=True)
plt.suptitle("Seaborn Plot with mplcursors Tooltips", y=1.02)
plt.show()

This method is highly effective for both basic plots and more complex visualizations created with libraries like Seaborn.

Solution 4: Using pick_event for Scatter Plots

This approach utilizes Matplotlib’s pick_event mechanism, which is triggered when a data point is “picked” (clicked or interacted with).

from matplotlib.pyplot import figure, show
import numpy as np

# Generate sample scatter data
x_data, y_data, colors_data, sizes_data = np.random.rand(4, 100)

def on_scatter_pick(event):
    """Callback function for scatter plot picking events."""
    indices = event.ind
    print('Scatter picked indices:', indices)
    print('Corresponding x:', np.take(x_data, indices))
    print('Corresponding y:', np.take(y_data, indices))

fig = figure()
ax_scatter = fig.add_subplot(111)
scatter_collection = ax_scatter.scatter(x_data, y_data, 100*sizes_data, colors_data, picker=True) # picker=True enables pick events

# Connect the pick event to the handler
fig.canvas.mpl_connect('pick_event', on_scatter_pick)

plt.title("Scatter Plot with Pick Event Handling")
show()

Solution 6: pick_event for Line Plots

Similar to the scatter plot example, this shows how to use pick_event for line plots.

import numpy as np
import matplotlib.pyplot as plt

fig, ax = plt.subplots()
ax.set_title('Click on points')

# Plot a line with a picker tolerance
line_artist, = ax.plot(np.random.rand(100), '-', picker=5)  # 5 points tolerance

def on_line_pick(event):
    """Callback function for line plot picking events."""
    this_line = event.artist
    x_data = this_line.get_xdata()
    y_data = this_line.get_ydata()
    indices = event.ind
    print('Picked points on line:', *zip(x_data[indices], y_data[indices]))

# Connect the pick event to the handler
fig.canvas.mpl_connect('pick_event', on_line_pick)

plt.show()

Solution 9: Multi-Line Annotation System

This solution provides a modular system for handling annotations on multiple lines, building upon existing examples.

import matplotlib.pyplot as plt

def update_annotation(ind_data, line_obj, annot_obj, y_data):
    """Updates the annotation with data from a line plot."""
    x_coords, y_coords = line_obj.get_data()
    annot_obj.xy = (x_coords[ind_data["ind"][0]], y_coords[ind_data["ind"][0]])
    x_values = " ".join(list(map(str, ind_data["ind"])))
    y_values = " ".join(str(y_data[n]) for n in ind_data["ind"])
    text_content = f"{x_values}, {y_values}"
    annot_obj.set_text(text_content)
    annot_obj.get_bbox_patch().set_alpha(0.4)

def hover_handler(event, line_info):
    """General hover handler for lines."""
    line, annot, y_data = line_info
    is_visible = annot.get_visible()
    if event.inaxes == ax:
        is_over_plot, current_data_indices = line.contains(event)
        if is_over_plot:
            update_annotation(current_data_indices, line, annot, y_data)
            annot.set_visible(True)
            fig.canvas.draw_idle()
        else:
            if is_visible:
                annot.set_visible(False)
                fig.canvas.draw_idle()

def plot_single_line_with_hover(x_vals, y_vals):
    """Plots a single line and sets up its hover annotation."""
    line_plot, = plt.plot(x_vals, y_vals, marker="o")
    annotation_obj = ax.annotate("", xy=(0, 0), xytext=(-20, 20), textcoords="offset points",
                        bbox=dict(boxstyle="round", fc="w"),
                        arrowprops=dict(arrowstyle="-&gt;"))
    annotation_obj.set_visible(False)
    line_details = [line_plot, annotation_obj, y_vals]
    fig.canvas.mpl_connect("motion_notify_event",
                           lambda event: hover_handler(event, line_details))

# Define data for multiple lines
x1_data = range(21)
y1_data = range(0, 21)
x2_data = range(21)
y2_data = range(0, 42, 2)

# Plot the lines
fig, ax = plt.subplots()
plot_single_line_with_hover(x1_data, y1_data)
plot_single_line_with_hover(x2_data, y2_data)
plt.title("Multi-Line Hover Annotations")
plt.show()

Solution 10: Status Bar Information Display

This approach leverages the Matplotlib status bar to display information about hovered artists without requiring extra libraries.

import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

class StatusbarHoverManager:
    """Manages hover information displayed in the Matplotlib status bar."""
    def __init__(self, ax):
        assert isinstance(ax, mpl.axes.Axes)

        # Initial hover handler to set default coordinate formatting
        def initial_hover(event):
            if event.inaxes != ax:
                return
            info = f'x={event.xdata:.2f}, y={event.ydata:.2f}'
            ax.format_coord = lambda x, y: info
        self.cid = ax.figure.canvas.mpl_connect("motion_notify_event", initial_hover)

        self.ax = ax
        self.artists = []
        self.labels = []

    def add_artist_labels(self, artist, label):
        """Adds an artist and its corresponding labels for hover tracking."""
        # Ensure artist is a single object if passed as a list
        if isinstance(artist, list):
            assert len(artist) == 1
            artist = artist[0]

        self.artists.append(artist)
        self.labels.append(label)

        # Reconnect hover handler to include new artist information
        def updated_hover(event):
            if event.inaxes != self.ax:
                return
            info = f'x={event.xdata:.2f}, y={event.ydata:.2f}'
            for i, current_artist in enumerate(self.artists):
                is_contained, details = current_artist.contains(event)
                if not is_contained:
                    continue
                indices = details.get('ind')
                if indices is not None:  # Artist contains multiple items
                    for idx in indices:
                        lbl = self.labels[i][idx]
                        info += f'; Artist [{i}, {idx}]: {lbl}'
                else: # Artist is a single item
                    lbl = self.labels[i]
                    info += f'; Artist [{i}]: {lbl}'
            self.ax.format_coord = lambda x, y: info

        self.ax.figure.canvas.mpl_disconnect(self.cid)
        self.cid = self.ax.figure.canvas.mpl_connect("motion_notify_event", updated_hover)

def demo_statusbar_hover():
    """Demonstrates the StatusbarHoverManager."""
    fig, ax = plt.subplots()
    hover_manager = StatusbarHoverManager(ax)

    # Add a polygon patch
    polygon_patch = mpl.patches.Polygon(
        [[0,0], [3, 5], [5, 4], [6,1]], closed=True, color='green', zorder=0)
    artist_patch = ax.add_patch(polygon_patch)
    hover_manager.add_artist_labels(artist_patch, 'polygon_label')

    # Add scatter points
    scatter_artist = ax.scatter([2.5, 1, 2, 3], [6, 1, 1, 7], c='blue', s=10**2)
    scatter_labels = ['point ' + str(ii) for ii in range(4)]
    hover_manager.add_artist_labels(scatter_artist, scatter_labels)

    # Add a line plot
    line_artist = ax.plot(
        [0, 0, 1, 5, 3], [0, 1, 1, 0, 2], marker='o', color='red')
    line_labels = ['segment ' + str(ii) for ii in range(5)]
    hover_manager.add_artist_labels(line_artist, line_labels)

    plt.title("Hover Info in Status Bar")
    plt.show()

if __name__ == "__main__":
    demo_statusbar_hover()

Solution 11: Enhanced Hover Functionality with Custom Labels

This solution builds upon existing examples to create a more modular hover system, allowing for custom labels and handling various artist types.

import matplotlib.pylab as plt
import numpy as np

plt.close('all')
fig_custom, ax_custom = plt.subplots()

# Generate sample data for multiple plots
hist_y, hist_x = np.histogram(np.random.randn(10000), bins=500)
hist_x = hist_x[:-1]
colors = ['#0000ff', '#00ff00','#ff0000']
x2_data, y2_data = hist_x, hist_y/10
x3_data, y3_data = hist_x, np.random.randn(500)*10+40

# Plot the data
h1_plot = ax_custom.plot(hist_x, hist_y, color=colors[0])[0] # Ensure it's the Line2D object
h2_plot = ax_custom.plot(x2_data, y2_data, color=colors[1])[0]
h3_scatter = ax_custom.scatter(x3_data, y3_data, color=colors[2], s=1)

all_artists = [h1_plot, h2_plot, h3_scatter]
all_labels = [list('ABCDE'*100), list('FGHIJ'*100), list('klmno'*100)] # Define custom labels

# Initialize annotation arrow
annotation = ax_custom.annotate("", xy=(0,0), xytext=(20,20),textcoords="offset points",
                    bbox=dict(boxstyle="round", fc="w"),
                    arrowprops=dict(arrowstyle="-&gt;"))
annotation.set_visible(False)

def on_custom_hover(event):
    """Handles hover events for multiple plot types."""
    if event.inaxes != ax_custom:
        return
    is_annotation_visible = annotation.get_visible()
    
    for i, artist in enumerate(all_artists):
        is_contained, detection_details = artist.contains(event)

        if is_contained:
            if('get_data' in dir(artist)): # For plot objects
                data_points = list(zip(*artist.get_data()))
            elif('get_offsets' in dir(artist)): # For scatter objects
                data_points = artist.get_offsets().data

            indices = detection_details['ind']
            
            # Set annotation properties
            annotation_position = data_points[indices[0]] # Use the first index found
            annotation.xy = annotation_position
            annotation.set_text(f'Pos={annotation_position}, Text={all_labels[i][indices[0]]}')
            annotation.get_bbox_patch().set_edgecolor(colors[i])
            annotation.get_bbox_patch().set_alpha(0.7)
            annotation.set_visible(True)
            fig_custom.canvas.draw_idle()
        else:
             if is_annotation_visible:
                 annotation.set_visible(False) # Hide annotation when not hovering
                 fig_custom.canvas.draw_idle()

fig_custom.canvas.mpl_connect('motion_notify_event', on_custom_hover)
plt.title("Enhanced Hover Functionality")
plt.show()

Solution 12: Patch and Class-Based Hovering

This approach encapsulates the hover logic within a class, making it reusable and organized, particularly for handling collections of patches.

import numpy as np
import matplotlib.pyplot as plt
import matplotlib.patches as mpatches
from matplotlib.collections import PatchCollection

np.random.seed(1)

class AnnotatedPatches:
    """Manages hover annotations for collections of Matplotlib patches."""
    def __init__(self, fig, ax):
        self.fig = fig
        self.ax = ax

        # Initialize annotation object
        self.annotation = self.ax.annotate("", xy=(0,0),
                            xytext=(20,20),
                            textcoords="offset points",
                            bbox=dict(boxstyle="round", fc="w"),
                            arrowprops=dict(arrowstyle="-&gt;"))
        self.annotation.set_visible(False)
        
        self.collections_data = {} # Stores patch collections
        self.coordinates_data = {} # Stores patch coordinates
        self.names_data = {} # Stores patch names
        self.is_active_data = {} # Tracks active highlighting

        # Connect the motion event callback
        self.motion_callback_id = self.fig.canvas.mpl_connect("motion_notify_event", self.handle_hover)

    def add_patches(self, group_name, patch_kind, color, xy_coords, names, *params):
        """Adds a collection of patches to the plot and hover manager."""
        if patch_kind == 'circle':
            patches = [mpatches.Circle(xy, *params, ec="none") for xy in xy_coords]
            collection = PatchCollection(patches, facecolor=color, alpha=0.5, edgecolor=None)
            self.ax.add_collection(collection)
        elif patch_kind == 'rect':
            patches = [mpatches.Rectangle(xy, *params, ec="none") for xy in xy_coords] 
            collection = PatchCollection(rectangles, facecolor=color, alpha=0.5, edgecolor=None)
            self.ax.add_collection(collection)
        else:
            raise ValueError('Unexpected patch kind', patch_kind)
            
        self.collections_data[group_name] = collection
        self.coordinates_data[group_name] = xy_coords
        self.names_data[group_name] = names
        self.is_active_data[group_name] = False
        
    def update_annotation(self, group_name, patch_indices):
        """Updates the annotation text and position for a specific patch."""
        self.annotation.xy = self.coordinates_data[group_name][patch_indices[0]]
        self.annotation.set_text(f'{group_name}: {self.names_data[group_name][patch_indices[0]]}')
        
        # Highlight the hovered patch by setting edge color
        self.collections_data[group_name].set_edgecolor('black')
        self.is_active_data[group_name] = True

    def handle_hover(self, event):
        """Handles the mouse motion event to detect hovering over patches."""
        is_annotation_visible = self.annotation.get_visible()
        updated_any_patch = False

        if event.inaxes == self.ax:            
            for group_name, collection in self.collections_data.items():
                is_contained, indices_dict = collection.contains(event)
                if is_contained:
                    self.update_annotation(group_name, indices_dict["ind"])
                    self.annotation.set_visible(True)
                    self.fig.canvas.draw_idle()
                    updated_any_patch = True
                else:
                    # Reset edge color if no longer hovering over this collection
                    if self.is_active_data[group_name]:
                        collection.set_edgecolor(None)
                        self.is_active_data[group_name] = False # Mark as inactive
                    
            # Hide annotation if mouse moved off all patches
            if (not updated_any_patch) and is_annotation_visible:
                self.annotation.set_visible(False)
                self.fig.canvas.draw_idle()

# Example Usage:
fig_patches, ax_patches = plt.subplots(tight_layout=True)

annotator = AnnotatedPatches(fig_patches, ax_patches)
# Add different types of patches with names and parameters
annotator.add_patches('Azure', 'circle', 'blue', np.random.uniform(0, 1, (4,2)), 'ABCD', 0.1)
annotator.add_patches('Lava', 'rect', 'red', np.random.uniform(0, 1, (3,2)), 'EFG', 0.1, 0.05)
annotator.add_patches('Emerald', 'rect', 'green', np.random.uniform(0, 1, (3,2)), 'HIJ', 0.05, 0.1)

plt.axis('equal') # Ensure aspect ratio is equal
plt.axis('off') # Hide axes
plt.title("Hovering Over Patches")
plt.show()
