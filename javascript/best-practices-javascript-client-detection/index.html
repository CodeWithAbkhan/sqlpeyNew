“Best Practices for Reliable JavaScript Client OS and Browser Detection”

Client detection via JavaScript relies heavily on inspecting properties available in the global navigator object. While historically dominated by parsing the userAgent string, modern approaches favor newer APIs like navigator.userAgentData while maintaining fallback mechanisms for broader compatibility. Reliability is key, especially when handling breaking changes like new Edge versions or misleading OS reports on Windows 11.

Solution 1: Advanced User Agent String Parsing with OS/Browser Specific Checks

One approach involves creating a self-contained script that exhaustively parses the userAgent string to extract details about the screen size, browser, OS, cookie status, and even Flash version (though Flash detection is becoming obsolete).

A critical update noted in this method addresses the detection of the modern, Chromium-based Microsoft Edge, distinguishing it from the legacy version.

Handling Windows 11 Inconsistencies

For accurate Windows version detection, particularly Windows 11 which may misleadingly report itself as Windows 10 in userAgent, the navigator.userAgentData API is leveraged. Since this is asynchronous, a timeout mechanism is necessary to ensure the UI updates correctly, although support is currently limited primarily to Edge, Chrome, and Opera.

Here is a foundational, extended client detection script:

/**
 * Comprehensive JavaScript Client Detection Utility
 * Incorporates legacy UA parsing and modern API fallbacks where possible.
 */
(function (window) {
    const unknown = '-';

    // 1. Screen Information
    let screenSize = '';
    if (screen.width) {
        const width = screen.width;
        const height = screen.height;
        screenSize = `${width} x ${height}`;
    }

    // 2. Browser Identification
    const nVer = navigator.appVersion;
    const nAgt = navigator.userAgent;
    let browser = navigator.appName;
    let version = `${parseFloat(nVer)}`;
    let majorVersion = parseInt(nVer, 10);
    let nameOffset, verOffset, ix;

    // Specific browser checks (order matters for precedence)
    if ((verOffset = nAgt.indexOf('YaBrowser')) !== -1) {
        browser = 'Yandex';
        version = nAgt.substring(verOffset + 10);
    } else if ((verOffset = nAgt.indexOf('SamsungBrowser')) !== -1) {
        browser = 'Samsung';
        version = nAgt.substring(verOffset + 15);
    } else if ((verOffset = nAgt.indexOf('UCBrowser')) !== -1) {
        browser = 'UC Browser';
        version = nAgt.substring(verOffset + 10);
    } else if ((verOffset = nAgt.indexOf('OPR')) !== -1) { // Opera Next/New Opera
        browser = 'Opera';
        version = nAgt.substring(verOffset + 4);
    } else if ((verOffset = nAgt.indexOf('Opera')) !== -1) { // Older Opera
        browser = 'Opera';
        version = nAgt.substring(verOffset + 6);
        if ((verOffset = nAgt.indexOf('Version')) !== -1) {
            version = nAgt.substring(verOffset + 8);
        }
    } else if ((verOffset = nAgt.indexOf('Edge')) !== -1) { // Legacy Edge
        browser = 'Microsoft Legacy Edge';
        version = nAgt.substring(verOffset + 5);
    } else if ((verOffset = nAgt.indexOf('Edg')) !== -1) { // Chromium Edge
        browser = 'Microsoft Edge';
        version = nAgt.substring(verOffset + 4);
    } else if ((verOffset = nAgt.indexOf('MSIE')) !== -1) { // IE
        browser = 'Microsoft Internet Explorer';
        version = nAgt.substring(verOffset + 5);
    } else if ((verOffset = nAgt.indexOf('Chrome')) !== -1) {
        browser = 'Chrome';
        version = nAgt.substring(verOffset + 7);
    } else if ((verOffset = nAgt.indexOf('Safari')) !== -1) {
        browser = 'Safari';
        version = nAgt.substring(verOffset + 7);
        if ((verOffset = nAgt.indexOf('Version')) !== -1) {
            version = nAgt.substring(verOffset + 8);
        }
    } else if ((verOffset = nAgt.indexOf('Firefox')) !== -1) {
        browser = 'Firefox';
        version = nAgt.substring(verOffset + 8);
    } else if (nAgt.indexOf('Trident/') !== -1) { // MSIE 11+
        browser = 'Microsoft Internet Explorer';
        version = nAgt.substring(nAgt.indexOf('rv:') + 3);
    } else if ((nameOffset = nAgt.lastIndexOf(' ') + 1) < (verOffset = nAgt.lastIndexOf('/'))) {
        browser = nAgt.substring(nameOffset, verOffset);
        version = nAgt.substring(verOffset + 1);
        if (browser.toLowerCase() === browser.toUpperCase()) {
            browser = navigator.appName;
        }
    }
    
    // Trim version string
    if ((ix = version.indexOf(';')) !== -1) version = version.substring(0, ix);
    if ((ix = version.indexOf(' ')) !== -1) version = version.substring(0, ix);
    if ((ix = version.indexOf(')')) !== -1) version = version.substring(0, ix);

    majorVersion = parseInt(`${version}`, 10);
    if (isNaN(majorVersion)) {
        version = `${parseFloat(nVer)}`;
        majorVersion = parseInt(nVer, 10);
    }

    // Mobile check
    const mobile = /Mobile|mini|Fennec|Android|iP(ad|od|hone)/.test(nVer);

    // Cookie check
    let cookieEnabled = (navigator.cookieEnabled) ? true : false;
    if (typeof navigator.cookieEnabled == 'undefined' && !cookieEnabled) {
        document.cookie = 'testcookie';
        cookieEnabled = (document.cookie.indexOf('testcookie') !== -1) ? true : false;
    }

    // 3. OS Identification (using clientStrings regex array)
    let os = unknown;
    const clientStrings = [
        {s:'Windows 10', r:/(Windows 10.0|Windows NT 10.0)/},
        {s:'Windows 8.1', r:/(Windows 8.1|Windows NT 6.3)/},
        {s:'Windows 8', r:/(Windows 8|Windows NT 6.2)/},
        {s:'Windows 7', r:/(Windows 7|Windows NT 6.1)/},
        {s:'Windows Vista', r:/Windows NT 6.0/},
        {s:'Windows Server 2003', r:/Windows NT 5.2/},
        {s:'Windows XP', r:/(Windows NT 5.1|Windows XP)/},
        {s:'Windows 2000', r:/(Windows NT 5.0|Windows 2000)/},
        {s:'Windows ME', r:/(Win 9x 4.90|Windows ME)/},
        {s:'Windows 98', r:/(Windows 98|Win98)/},
        {s:'Windows 95', r:/(Windows 95|Win95|Windows_95)/},
        {s:'Windows NT 4.0', r:/(Windows NT 4.0|WinNT4.0|WinNT|Windows NT)/},
        {s:'Windows CE', r:/Windows CE/},
        {s:'Windows 3.11', r:/Win16/},
        {s:'Android', r:/Android/},
        {s:'Open BSD', r:/OpenBSD/},
        {s:'Sun OS', r:/SunOS/},
        {s:'Chrome OS', r:/CrOS/},
        {s:'Linux', r:/(Linux|X11(?!.*CrOS))/},
        {s:'iOS', r:/(iPhone|iPad|iPod)/},
        {s:'Mac OS X', r:/Mac OS X/},
        {s:'Mac OS', r:/(Mac OS|MacPPC|MacIntel|Mac_PowerPC|Macintosh)/},
        {s:'QNX', r:/QNX/},
        {s:'UNIX', r:/UNIX/},
        {s:'BeOS', r:/BeOS/},
        {s:'OS/2', r:/OS\/2/},
        {s:'Search Bot', r:/(nuhk|Googlebot|Yammybot|Openbot|Slurp|MSNBot|Ask Jeeves\/Teoma|ia_archiver)/}
    ];
    
    for (const cs of clientStrings) {
        if (cs.r.test(nAgt)) {
            os = cs.s;
            break;
        }
    }

    let osVersion = unknown;

    if (/Windows/.test(os)) {
        osVersion = /Windows (.*)/.exec(os)[1] || unknown; // Extract version if present
        os = 'Windows';
    }

    switch (os) {
        case 'Mac OS X':
            const macOsXMatch = /(?:Mac OS X) ([\.\_\d]+)/.exec(nAgt);
            osVersion = macOsXMatch ? macOsXMatch[1].replace(/_/g, '.') : unknown;
            break;
        case 'Android':
            const androidMatch = /Android ([\.\_\d]+)/.exec(nAgt);
            osVersion = androidMatch ? androidMatch[1] : unknown;
            break;
        case 'iOS':
            const iosMatch = /OS (\d+)_(\d+)_?(\d+)?/.exec(nVer);
            if (iosMatch) {
                osVersion = `${iosMatch[1]}.${iosMatch[2]}.${iosMatch[3] || 0}`;
            }
            break;
    }
    
    // Flash detection (largely historical now)
    let flashVersion = 'no check';
    // ... (Flash detection logic omitted for brevity, as per original structure)

    window.jscd = {
        screen: screenSize,
        browser: browser,
        browserVersion: version,
        browserMajorVersion: majorVersion,
        mobile: mobile,
        os: os,
        osVersion: osVersion,
        cookies: cookieEnabled,
        flashVersion: flashVersion
    };
}(this));

Solution 7: OS Detection based on navigator.platform Array Check

This provides a cleaner, array-based check against navigator.platform, focusing primarily on OS identification rather than browser versioning.

function getOS() {
  const userAgent = window.navigator.userAgent;
  const platform = window.navigator.platform;
  const macosPlatforms = ['Macintosh', 'MacIntel', 'MacPPC', 'Mac68K'];
  const windowsPlatforms = ['Win32', 'Win64', 'Windows', 'WinCE'];
  const iosPlatforms = ['iPhone', 'iPad', 'iPod'];
  let os = null;

  if (macosPlatforms.indexOf(platform) !== -1) {
    os = 'Mac OS';
  } else if (iosPlatforms.indexOf(platform) !== -1) {
    os = 'iOS';
  } else if (windowsPlatforms.indexOf(platform) !== -1) {
    os = 'Windows';
  } else if (/Android/.test(userAgent)) {
    os = 'Android';
  } else if (!os && /Linux/.test(platform)) {
    os = 'Linux';
  }
  return os;
}
console.log(`Detected OS (platform array): ${getOS()}`);

Solution 8: Hybrid Approach using userAgentData (Modern API)

This superior modern approach prioritizes the User-Agent Client Hints API (navigator.userAgentData), which provides structured data, falling back to the deprecated navigator.platform if the API is unavailable. This requires a secure context (HTTPS).

async function getPlatform() {
    // Modern, structured detection (requires HTTPS)
    if (typeof navigator.userAgentData !== 'undefined' && navigator.userAgentData !== null) {
        // Request high-entropy values asynchronously
        try {
            const data = await navigator.userAgentData.getHighEntropyValues([
                "platform", 
                "platformVersion"
            ]);
            return data.platform;
        } catch (error) {
            // Fallback if high-entropy values fail or permission denied
            console.warn("UserAgentData high-entropy failed, falling back:", error);
        }
    }
    
    // Deprecated fallback
    if (typeof navigator.platform !== 'undefined') {
        // Specific adjustment for Android reports on platform
        if (typeof navigator.userAgent !== 'undefined' && /android/.test(navigator.userAgent.toLowerCase())) {
            return 'android';
        }
        return navigator.platform;
    }
    return 'unknown';
}

getPlatform().then(platform => {
    console.log(`Hybrid Platform: ${platform}`);

    console.log(`Is this Windows? ${isWindows}`);
});

FAQs on JavaScript Client OS and Browser Detection

Q: Which method for client detection is most reliable today

ANS: The most reliable method currently involves using navigator.userAgentData for structured data, especially for OS versioning, while implementing robust fallback logic using classic User Agent string parsing (like Solution 1 or 8) for compatibility with older browsers that lack User Agent Client Hints support.

Q: Why is navigator platform deprecated

ANS: navigator.platform is deprecated because the value it returns is often easily spoofed by users or security settings, leading to inaccurate client profiling. Modern specs prefer the structured, opt-in approach of navigator.userAgentData.

Q: Can I get the exact Windows version from the User Agent reliably

ANS: Directly parsing the User Agent string for exact Windows versions is brittle. Modern operating systems (like Windows 11) frequently report older versions in the UA. Relying on navigator.userAgentData.platformVersion (if available) is preferable.

Q: What is the breaking change with Microsoft Edge

ANS: The breaking change involved Microsoft replacing the old Edge rendering engine with Chromium. This required detection logic to look for ‘Edg/’ (Chromium Edge) instead of just ‘Edge’ to correctly identify the newer browser version.
