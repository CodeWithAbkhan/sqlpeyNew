“Convert Unix Timestamp to Readable Time in JavaScript”

A Unix timestamp, representing the number of seconds that have elapsed since the Unix Epoch (January 1, 1970, at 00:00:00 Coordinated Universal Time (UTC)), often needs to be displayed in a more understandable format for users. JavaScript’s built-in Date object provides several ways to achieve this, from simple time displays to full date and time strings. Libraries like Moment.js and Day.js offer more advanced formatting capabilities.

Let’s explore various methods to tackle this conversion:

Basic Time Formatting (HH:MM:SS)

JavaScript’s Date object requires milliseconds, so the Unix timestamp (in seconds) must be multiplied by 1000.

Solution 1: Direct Date Object Manipulation

This approach directly extracts hours, minutes, and seconds, ensuring two digits for each component by padding with a leading zero if necessary.

let unix_timestamp = 1549312452;

// Create a new JavaScript Date object based on the timestamp
// multiplied by 1000 for milliseconds
var date = new Date(unix_timestamp * 1000);

// Extract hours, minutes, and seconds
var hours = date.getHours();
var minutes = "0" + date.getMinutes();
var seconds = "0" + date.getSeconds();

// Combine to form HH:MM:SS format
var formattedTime = hours + ':' + minutes.substr(-2) + ':' + seconds.substr(-2);

console.log(formattedTime); // Example output: 20:34:12

Solution 5: toISOString() for Precise Formatting

The toISOString() method provides a standardized, extended ISO 8601 format. We can then slice this string to extract the time component.

function format_time(s) {
  // Convert seconds to milliseconds and create a Date object
  // then use toISOString() and slice to get HH:MM:SS
  return new Date(s * 1e3).toISOString().slice(-13, -5);
}

console.log(format_time(12345)); // Example output: 03:25:45

Solution 7: toISOString() with replace()

A variation of the above, this method uses replace() to substitute the ‘T’ separator with a space for a more common date-time appearance.

console.log(new Date(1549312452 * 1000).toISOString().slice(0, 19).replace('T', ' '));
// Example output: 2019-02-04 20:34:12

Solution 19: Padding for Accuracy

This solution addresses potential issues where single-digit hours, minutes, or seconds might not be zero-padded correctly.

var timestamp = 1439329773; // Replace with your timestamp
var date = new Date(timestamp * 1000);
// Format date with zero-padding for day, month, hours, and minutes
var formattedDate = ('0' + date.getDate()).slice(-2) + '/' + ('0' + (date.getMonth() + 1)).slice(-2) + '/' + date.getFullYear() + ' ' + ('0' + date.getHours()).slice(-2) + ':' + ('0' + date.getMinutes()).slice(-2);
console.log(formattedDate); // Example output: 12/08/2015 00:49

Solution 21: Flexible Separator Function

This function allows specifying a custom separator for the time components and handles zero-padding.

var convertTime = function(timestamp, separator) {
    var pad = function(input) {return input < 10 ? "0" + input : input;};
    var date = timestamp ? new Date(timestamp * 1000) : new Date(); // Use provided timestamp or current date
    return [
        pad(date.getHours()),
        pad(date.getMinutes()),
        pad(date.getSeconds())
    ].join(typeof separator !== 'undefined' ?  separator : ':' ); // Default separator is ':'
}

console.log(convertTime(1061351153)); // Output: 05:45:53
console.log(convertTime(920535115, '/')); // Output: 09/11/55

Solution 23: Custom twoDigits Helper

A dedicated helper function ensures that numbers less than 10 are prefixed with a ‘0’.

// Format value as two digits: 0 => 00, 1 => 01
function twoDigits(value) {
   if(value < 10) {
    return '0' + value;
   }
   return value;
}

var date = new Date(unix_timestamp*1000);
// display in format HH:MM:SS
var formattedTime = twoDigits(date.getHours())
      + ':' + twoDigits(date.getMinutes())
      + ':' + twoDigits(date.getSeconds());
console.log(formattedTime);

Full Date and Time Formatting

For displaying the date along with the time, several methods are available.

Solution 2: Detailed Date and Time Conversion

This function returns a string including the date, month, year, and time.

function timeConverter(UNIX_timestamp){
  var a = new Date(UNIX_timestamp * 1000);
  var months = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  var year = a.getFullYear();
  var month = months[a.getMonth()];
  var date = a.getDate();
  var hour = a.getHours();
  var min = a.getMinutes();
  var sec = a.getSeconds();
  var time = date + ' ' + month + ' ' + year + ' ' + hour + ':' + min + ':' + sec ;
  return time;
}
console.log(timeConverter(0)); // Example output: 1 Jan 1970 00:00:00

console.log(s) // Example output: 8/30/2017

var s = new Date(1504095567183).toLocaleTimeString("en-US")
console.log(s) // Example output: 3:19:27 PM

Solution 10: Accessing Various Date Components

This demonstrates accessing different parts of the Date object, including the full date string and individual components.

var date = new Date(+d); // Ensure it's treated as a number

Solution 15: Relative Time Formatting

This advanced solution formats the date relative to the current day, showing “today,” “yesterday,” or the date with month and year.

function timeConverter(t) {
    var a = new Date(t * 1000);
    var today = new Date();
    var yesterday = new Date(Date.now() - 86400000); // 24 hours in milliseconds
    var months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
    var year = a.getFullYear();
    var month = months[a.getMonth()];
    var date = a.getDate();
    var hour = a.getHours();
    var min = a.getMinutes();

    // Compare dates by setting time to 00:00:00 for accurate day comparison
    if (a.setHours(0,0,0,0) == today.setHours(0,0,0,0))
        return 'today, ' + hour + ':' + min;
    else if (a.setHours(0,0,0,0) == yesterday.setHours(0,0,0,0))
        return 'yesterday, ' + hour + ':' + min;
    else if (year == today.getFullYear())
        return date + ' ' + month + ', ' + hour + ':' + min; // Same year, different day
    else
        return date + ' ' + month + ' ' + year + ', ' + hour + ':' + min; // Different year
}

Solution 16: YYYY-MM-DD HH:MM:SS Format

This function creates a timestamp string in a commonly used format.

function getTIMESTAMP() {
  var date = new Date();
  var year = date.getFullYear();
  // Pad month, day, hour, minute, second with leading zeros if needed
  var month = ("0" + (date.getMonth() + 1)).slice(-2);
  var day = ("0" + date.getDate()).slice(-2);
  var hour = ("0" + date.getHours()).slice(-2);
  var minutes = ("0" + date.getMinutes()).slice(-2);
  var seconds = ("0" + date.getSeconds()).slice(-2);

  return year + "-" + month + "-" + day + " " + hour + ":" + minutes + ":" + seconds;
}
// Example output: 2024-07-22 10:30:01

This method leverages toLocaleString() and then manipulates the resulting string.

const result = (dateTimeStr.split(", ")[1]).split(":").join("/"); // Formats time as HH/MM/SS
console.log(result);

Solution 20: getMonth() + 1 for Correct Month Display

This function ensures the month is displayed correctly by adding 1 to the result of getMonth().

function getDateTimeFromTimestamp(unixTimeStamp) {
    let date = new Date(unixTimeStamp); // Assumes timestamp is in milliseconds
    return ('0' + date.getDate()).slice(-2) + '/' + ('0' + (date.getMonth() + 1)).slice(-2) + '/' + date.getFullYear() + ' ' + ('0' + date.getHours()).slice(-2) + ':' + ('0' + date.getMinutes()).slice(-2);
}

const myTime = getDateTimeFromTimestamp(1435986900000);
console.log(myTime); // Example output: 01/05/2000 11:00 (if timestamp was 1435986900000)

Utilizing Libraries

For more complex date manipulation and formatting, libraries are highly recommended.

Solution 8 & 9 & 25: Moment.js

Moment.js is a popular library for date and time manipulation in JavaScript.

// Make sure to install Moment.js: npm install moment
const moment = require('moment');

// Using a Unix timestamp (seconds)
var timestamp = moment.unix(1293683278);
console.log( timestamp.format("HH/mm/ss") ); // Example output: 10:28:00

// Formatting with date
const dateTimeString = moment.unix(1466760000).format("DD-MM-YYYY HH:mm:ss");
console.log(dateTimeString); // Example output: 24-06-2016 10:00:00

// Formatting current date and time
console.log( moment().format('YYYY-MM-DD hh:mm:ss') ); // Example output: 2024-07-22 11:55:43

Solution 17: Day.js

Day.js is a lighter alternative to Moment.js, offering similar functionality.

// Make sure to install Day.js: npm install dayjs
const dayjs = require('dayjs');

// Using a timestamp in milliseconds
const result = dayjs(1504052527183).format("HH/mm/ss");
console.log(result); // Example output: 09/42/07

Shortest Solutions

For concise code, especially in environments where brevity is key:

Solution 26: String Slicing

This is a very compact method to extract the time portion.

let ts = 1549312452;
let time = (new Date(ts*1000)+'').slice(16,24); // Extracts HH:MM:SS from the default date string representation

console.log(time); // Example output: 20:34:12

Handling Durations

If the timestamp represents a duration rather than a point in time:

Solution 29: Calculating Hours, Minutes, Seconds from Duration

This approach converts a total number of seconds into a standard duration format.

var timestamp = 3661; // Example: 1 hour, 1 minute, 1 second

var hours = Math.floor(timestamp / 60 / 60);
var minutes = Math.floor((timestamp - hours * 60 * 60) / 60);
var seconds = Math.floor(timestamp - hours * 60 * 60 - minutes * 60 );
var duration = hours + ':' + minutes + ':' + seconds;

console.log(duration); // Example output: 1:1:1

FAQs on Convert Unix Timestamp to Readable Time in JavaScript

Q: What is a Unix timestamp

ANS: A Unix timestamp is the number of seconds that have elapsed since January 1, 1970, at 00:00:00 Coordinated Universal Time (UTC), commonly known as the Unix Epoch.

Q: Why do I need to multiply the Unix timestamp by 1000

ANS: JavaScript’s Date object works with milliseconds, while Unix timestamps are typically in seconds. Multiplying by 1000 converts seconds to milliseconds, which is required for the Date constructor.

Q: Which JavaScript method is best for formatting dates

ANS: For simple formatting, native Date methods like getHours, getMinutes, getSeconds, and toLocaleTimeString are effective. For more advanced or localized formatting, Intl.DateTimeFormat is recommended. Libraries like Moment.js or Day.js offer extensive formatting options and are useful for complex date manipulations.

Q: How can I ensure hours, minutes, and seconds are always two digits

ANS: You can use string manipulation methods like slice(-2) after padding with a leading zero ("0" + value) or use helper functions like padStart(2, '0') or custom twoDigits functions to ensure values like ‘5’ become ‘05’.

Q: Can I format the time based on the user's local settings

ANS: Yes, methods like toLocaleTimeString(), toLocaleDateString(), and Intl.DateTimeFormat(undefined, options) automatically format dates and times according to the user’s browser or system locale settings.
