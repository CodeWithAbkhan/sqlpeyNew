“Generate Random Hex Colors in JavaScript: Effective Techniques”

Encountering the need to dynamically generate colors for web elements? Whether it’s for interactive charts, random background styles, or unique UI elements, JavaScript offers numerous ways to create random color codes. This exploration delves into various effective techniques, ranging from concise one-liners to more controlled methods for generating distinct palettes.

Let’s examine some of the most popular and practical approaches.

Solution 3: Generating “Evenly Spaced” Colors (Rainbow Function)

If you need colors that are visually distinct and don’t cluster together, this rainbow function is ideal, especially for applications like Google Maps markers.

/**
 * Generates vibrant, "evenly spaced" colors.
 * @param {number} numOfSteps - Total number of colors to generate.
 * @param {number} step - The current color's index.
 * @returns {string} A hex color string.
 */
function rainbow(numOfSteps, step) {
    let r, g, b;
    const h = step / numOfSteps;
    const i = ~~(h * 6);
    const f = h * 6 - i;
    const q = 1 - f;
    switch (i % 6) {
        case 0: [r, g, b] = [1, f, 0]; break;
        case 1: [r, g, b] = [q, 1, 0]; break;
        case 2: [r, g, b] = [0, 1, f]; break;
        case 3: [r, g, b] = [0, q, 1]; break;
        case 4: [r, g, b] = [f, 0, 1]; break;
        case 5: [r, g, b] = [1, 0, q]; break;
    }
    const c = "#" +
        ("00" + ~~(r * 255)).slice(-2) +
        ("00" + ~~(g * 255)).slice(-2) +
        ("00" + ~~(b * 255)).slice(-2);
    return c;
}

You can see this in action with a “Simple JavaScript Rainbow Color Generator for Google Map Markers.”

}

Solution 9: Controlling Brightness with RGB

function getRandColor(brightness) {
    // Six levels of brightness from 0 to 5, 0 being the darkest
    const rgb = [Math.random() * 256, Math.random() * 256, Math.random() * 256];
    const mix = [brightness * 51, brightness * 51, brightness * 51]; // 51 => 255/5
    const mixedrgb = rgb.map((val, i) => Math.round((val + mix[i]) / 2.0));
    return "rgb(" + mixedrgb.join(",") + ")";
}

Solution 12: Versatile random_color Function

This function offers flexibility by allowing you to specify the output format (hex, rgb, rgba).

function random_color(format) {
    let rint = Math.round(0xffffff * Math.random());
    switch (format) {
        case 'hex':
            return ('#0' + rint.toString(16)).replace(/^#0([0-9a-f]{6})$/i, '#$1');
        case 'rgb':
            return `rgb(${(rint >> 16)}, ${(rint >> 8) & 255},${rint & 255})`;
        default:
            return rint;
    }
}

    ("0" + parseInt(rgb[1], 10).toString(16)).slice(-2) +

// HSV values in [0..1[
// returns [r, g, b] values from 0 to 255
const hsv_to_rgb = (h, s, v) => {
  const h_i = Math.floor(h * 6);
  const f = h * 6 - h_i;
  const p = v * (1 - s);
  const q = v * (1 - (f * s));
  const t = v * (1 - (1 - f) * s);
  let r, g, b;
  switch (h_i) {
    case(0): [r, g, b] = [v, t, p]; break;
    case(1): [r, g, b] = [q, v, p]; break;
    case(2): [r, g, b] = [p, v, t]; break;
    case(3): [r, g, b] = [p, q, v]; break;
    case(4): [r, g, b] = [t, p, v]; break;
    case(5): [r, g, b] = [v, p, q]; break;
  }
  return [Math.floor(r * 256), Math.floor(g * 256), Math.floor(b * 256)];
}

  return colorArray;
}

Solution 20: Controlled RGB Ranges

This function provides fine-grained control over color generation by allowing you to specify ranges for the Red, Green, and Blue components.

function generateColor(ranges) {
    if (!ranges) {
        ranges = [
            [150, 256], // Red component range
            [0, 190],   // Green component range
            [0, 30]     // Blue component range
        ];
    }
    const g = function() {
        // Select a random range and remove it
        const range = ranges.splice(Math.floor(Math.random() * ranges.length), 1)[0];
        // Pick a random number from within the range
        return Math.floor(Math.random() * (range[1] - range[0])) + range[0];
    };
    return `rgb(${g()},${g()},${g()})`;
}

Solution 29: Distinct HSV Palette Generation

This advanced function generates a palette of HSV colors, prioritizing distinctness by calculating Euclidean distances in the HSV cone. It allows range restrictions for hue, saturation, and value.

  /**
   * Generates a random palette of HSV colors. Attempts to pick colors
   * that are as distinct as possible within the desired HSV range.
   *
   * @param {object}    [options]
   * @param {number}    [options.numColors=10] - number of colors to generate
   * @param {number[]}  [options.hRange=[0,1]] - hue range
   * @param {number[]}  [options.sRange=[0,1]] - saturation range
   * @param {number[]}  [options.vRange=[0,1]] - value range
   * @param {number[][]} [options.exclude=[[0,0,0],[0,0,1]]] - colors to exclude
   *
   * @returns {number[][]} array of HSV colors ([hue, saturation, value])
   */
  function randomHSVPalette(options) {
    function random(min, max) {
      return min + Math.random() * (max - min);
    }

    function HSVtoXYZ(hsv) {
      const h = hsv[0]; const s = hsv[1]; const v = hsv[2];
      const angle = h * Math.PI * 2;
      return [Math.sin(angle) * s * v, Math.cos(angle) * s * v, v];
    }

    function distSq(a, b) {
      const dx = a[0] - b[0]; const dy = a[1] - b[1]; const dz = a[2] - b[2];
      return dx * dx + dy * dy + dz * dz;
    }

    options = options || {};
    const numColors = options.numColors || 10;
    const hRange = options.hRange || [0, 1];
    const sRange = options.sRange || [0, 1];
    const vRange = options.vRange || [0, 1];
    const exclude = options.exclude || [[0, 0, 0], [0, 0, 1]];

    const points = exclude.map(HSVtoXYZ);
    const result = [];

    while (result.length < numColors) {
      let bestHSV, bestXYZ, bestDist = 0;
      for (let i = 0; i < 20; i++) {
        const hsv = [random(hRange[0], hRange[1]), random(sRange[0], sRange[1]), random(vRange[0], vRange[1])];
        const xyz = HSVtoXYZ(hsv);
        let minDist = 10;
        points.forEach(point => minDist = Math.min(minDist, distSq(xyz, point)));
        if (minDist > bestDist) {
          bestHSV = hsv; bestXYZ = xyz; bestDist = minDist;
        }
      }
      points.push(bestXYZ);
      result.push(bestHSV);
    }
    return result;
  }

  function HSVtoRGB(hsv) {
    const h = hsv[0]; const s = hsv[1]; const v = hsv[2];
    const i = ~~(h * 6); const f = h * 6 - i;
    const p = v * (1 - s); const q = v * (1 - f * s); const t = v * (1 - (1 - f) * s);
    let v_rgb = ~~(255 * v), p_rgb = ~~(255 * p), q_rgb = ~~(255 * q), t_rgb = ~~(255 * t);
    switch (i % 6) {
      case 0: return [v_rgb, t_rgb, p_rgb];
      case 1: return [q_rgb, v_rgb, p_rgb];
      case 2: return [p_rgb, v_rgb, t_rgb];
      case 3: return [p_rgb, q_rgb, v_rgb];
      case 4: return [t_rgb, p_rgb, v_rgb];
      case 5: return [v_rgb, p_rgb, q_rgb];
    }
  }

  function RGBtoCSS(rgb) {
    const r = rgb[0]; const g = rgb[1]; const b = rgb[2];
    const rgbVal = (r << 16) + (g << 8) + b;
    return '#' + ('000000' + rgbVal.toString(16)).slice(-6);
  }
