“JavaScript: Convert String to Title Case Effectively”

There are numerous approaches to achieve title case conversion in JavaScript. The choice often depends on the specific requirements, such as handling acronyms, minor words, or special characters. Here are several effective methods:

Exploring Title Case Conversion Techniques in JavaScript

Solution 1: Basic Regex Replacement

This method uses a regular expression to find sequences of word characters (\w\S*) and applies capitalization to the first character while converting the rest to lowercase.

function toTitleCase(str) {
  return str.replace(
    /\w\S*/g,
    text => text.charAt(0).toUpperCase() + text.substring(1).toLowerCase()
  );
}

const example = 'john smith';
console.log(`"${example}" becomes "${toTitleCase(example)}"`);

Solution 2: CSS text-transform: capitalize

For client-side rendering, CSS offers a straightforward approach. However, it has limitations in accurately handling apostrophes or hyphenated words.

text-transform: capitalize;

Be aware of its behavior:

  • hello world becomes Hello World
  • HELLO WORLD remains HELLO WORLD
  • emily-jane o'brien becomes Emily-jane O'brien (incorrect)
  • Maria von Trapp becomes Maria Von Trapp (incorrect)

Solution 3: Prototype Extension with Regex

This approach extends the String prototype for a more object-oriented feel, adapting a well-known function.

String.prototype.toProperCase = function () {
    return this.replace(/\w\S*/g, function(txt){
        return txt.charAt(0).toUpperCase() + txt.substr(1).toLowerCase();
    });
};

console.log("pascal".toProperCase()); // "Pascal"

Solution 4: split, map, and join Method

A clean and readable method involves splitting the string into an array of words, mapping each word to its capitalized form, and then joining them back.

const str = "foo bar baz";
const newStr = str.split(' ')
   .map(w => w[0].toUpperCase() + w.substring(1).toLowerCase())
   .join(' ');
console.log(newStr); // "Foo Bar Baz"

Solution 5: Preserving Acronyms and Minor Words

For more sophisticated title casing, consider a function that handles specific rules for minor words (like articles and prepositions) and preserves uppercase acronyms.

String.prototype.toTitleCase = function() {
  let str = this.replace(/([^\W_]+[^\s-]*) */g, function(txt) {
    return txt.charAt(0).toUpperCase() + txt.substr(1).toLowerCase();
  });

  const lowers = ['A', 'An', 'The', 'And', 'But', 'Or', 'For', 'Nor', 'As', 'At', 'By', 'From', 'In', 'Into', 'Near', 'Of', 'On', 'Onto', 'To', 'With'];
  for (let i = 0; i < lowers.length; i++) {
    str = str.replace(new RegExp('\\s' + lowers[i] + '\\s', 'g'), function(txt) {
      return txt.toLowerCase();
    });
  }

  const uppers = ['Id', 'Tv'];
  for (let i = 0; i < uppers.length; i++) {
    str = str.replace(new RegExp('\\b' + uppers[i] + '\\b', 'g'), uppers[i].toUpperCase());
  }

  return str;
}

console.log("TO LOGIN TO THIS SITE and watch tv, please enter a valid id:".toTitleCase());
// Returns: "To Login to This Site and Watch TV, Please Enter a Valid ID:"

Solution 6: Lowercase then Uppercase First Letter

This concise approach first converts the entire string to lowercase and then uses a regex to uppercase the first letter of each word boundary.

function titleCase(str) {
  return str.toLowerCase().replace(/\b\w/g, s => s.toUpperCase());
}

console.log(titleCase('iron man')); // "Iron Man"
console.log(titleCase('iNcrEdible hulK')); // "Incredible Hulk"

Solution 8: Performance Benchmarks and for Loop

For performance-critical scenarios, a traditional for loop might outperform regex-heavy solutions, though modern JavaScript engines often optimize regex well. Benchmarks can vary significantly across environments.

function titleize(str) {
    let upper = true;
    let newStr = "";
    for (let i = 0, l = str.length; i < l; i++) {
        if (str[i] === " ") {
            upper = true;
            newStr += str[i];
            continue;
        }
        newStr += upper ? str[i].toUpperCase() : str[i].toLowerCase();
        upper = false;
    }
    return newStr;
}

const testStr = "the QUICK BrOWn Fox jUMPS oVeR the LAzy doG";
console.log(titleize(testStr)); // "The Quick Brown Fox Jumps Over The Lazy Dog"

Solution 11: No Regex Approach

For those who prefer to avoid regex, this method splits, iterates, capitalizes, and joins.

String.prototype.toProperCase = function() {
  var words = this.split(' ');
  var results = [];
  for (var i = 0; i < words.length; i++) {
    var letter = words[i].charAt(0).toUpperCase();
    results.push(letter + words[i].slice(1));
  }
  return results.join(' ');
};

console.log('john smith'.toProperCase()); // "John Smith"

Solution 12: Handling “Glue” Words

This advanced solution allows specifying words that should remain lowercase unless they are the first or last word in the string.

/**
 * @param String str The text to be converted to titleCase.
 * @param Array glue the words to leave in lowercase.
 */
var titleCase = function(str, glue){
    glue = (glue) ? glue : ['of', 'for', 'and'];
    return str.replace(/(\w)(\w*)/g, function(_, i, r){
        var j = i.toUpperCase() + (r != null ? r : "");
        return (glue.indexOf(j.toLowerCase())<0)?j:j.toLowerCase();
    });
};

// Handling leading glue words
var titleCaseWithLeading = function(str, glue){
    glue = !!glue ? glue : ['of', 'for', 'and', 'a'];
    var first = true;
    return str.replace(/(\w)(\w*)/g, function(_, i, r) {
        var j = i.toUpperCase() + (r != null ? r : '').toLowerCase();
        var result = ((glue.indexOf(j.toLowerCase()) < 0) || first) ? j : j.toLowerCase();
        first = false;
        return result;
    });
};

console.log(titleCase("the lord of the rings")); // "The Lord of the Rings"
console.log(titleCaseWithLeading("a tale of two cities")); // "A Tale of Two Cities"

Solution 13: Grammatical Title Case

This sophisticated approach considers grammatical rules, including prepositions, articles, and conjunctions, for more editorial-style titles. It also handles whitespace consolidation and special character replacement.

const toTitleCase = (str) => {
  const articles = ['a', 'an', 'the'];
  const conjunctions = ['for', 'and', 'nor', 'but', 'or', 'yet', 'so'];
  const prepositions = [
    'with', 'at', 'from', 'into','upon', 'of', 'to', 'in', 'for',
    'on', 'by', 'like', 'over', 'plus', 'but', 'up', 'down', 'off', 'near'
  ];

  const replaceCharsWithSpace = (str) => str.replace(/[^0-9a-z&amp;/\\]/gi, ' ').replace(/(\s\s+)/gi, ' ');
  const capitalizeFirstLetter = (str) => str.charAt(0).toUpperCase() + str.substr(1);
  const normalizeStr = (str) => str.toLowerCase().trim();
  const shouldCapitalize = (word, fullWordList, posWithinStr) => {
    if ((posWithinStr === 0) || (posWithinStr === fullWordList.length - 1)) {
      return true;
    }
    return !(articles.includes(word) || conjunctions.includes(word) || prepositions.includes(word));
  }

  str = replaceCharsWithSpace(str);
  str = normalizeStr(str);

  let words = str.split(' ');
  if (words.length <= 2) {
    words = words.map(w => capitalizeFirstLetter(w));
  }
  else {
    for (let i = 0; i < words.length; i++) {
      words[i] = (shouldCapitalize(words[i], words, i) ? capitalizeFirstLetter(words[i], words, i) : words[i]);
    }
  }

  return words.join(' ');
}

Solution 15: Handling Specific Name Formats

This function attempts to handle names like “McDonald” or “O’Toole” by applying specific capitalization rules.

function toProperCase(s) {
  return s.toLowerCase().replace( /\b((m)(a?c))?(\w)/g,
          function($1, $2, $3, $4, $5) { if($2){return $3.toUpperCase()+$4+$5.toUpperCase();} return $1.toUpperCase(); });
}

console.log(toProperCase("mcdonald o'toole")); // "Mcdonald O'Toole"

Solution 17: Array map with Split and toLowerCase

Ensures consistent capitalization by first converting the string to lowercase.

function titleCase(str) {
  str = str.toLowerCase(); // ensure the HeLlo will become Hello at the end
  var words = str.split(" ");

  var capitalized = words.map(function(word) {
    return word.charAt(0).toUpperCase() + word.substring(1, word.length);
  });
  return capitalized.join(" ");
}

console.log(titleCase("I'm a little tea pot")); // "I'm A Little Tea Pot"

For cases with more complex separators (like / or -), a more robust regex can be used:

const capitalize = str => str.length
  ? str[0].toUpperCase() +
    str.slice(1).toLowerCase()
  : '';

const escape = str => str.replace(/./g, c => `\\${c}`);
const titleCaseWithSeparators = (sentence, seps = ' _-/') => {
  let wordPattern = new RegExp(`[^${escape(seps)}]+`, 'g');
  return sentence.replace(wordPattern, capitalize);
};

console.log( titleCaseWithSeparators("I'm a little/small tea pot.") ); // "I'm A Little/Small Tea Pot."

Solution 20: Unicode Support for Accented Characters

Solution 26: Language-Aware Title Casing with Exceptions

This advanced function handles language-specific nuances (like French articles) and allows toggling exceptions for minor words.

String.prototype.titlecase = function(lang, withLowers = false) {
    let string = this.replace(/([^\s:\-'])([^\s:\-']*)/g, function(txt) {
        return txt.charAt(0).toUpperCase() + txt.substr(1).toLowerCase();
    }).replace(/Mc(.)/g, function(match, next) {
        return 'Mc' + next.toUpperCase();
    });

    if (withLowers) {
        const lowers = lang === 'EN'
            ? ['A', 'An', 'The', 'At', 'By', 'For', 'In', 'Of', 'On', 'To', 'Up', 'And', 'As', 'But', 'Or', 'Nor', 'Not']
            : ['Un', 'Une', 'Le', 'La', 'Les', 'Du', 'De', 'Des', 'À', 'Au', 'Aux', 'Par', 'Pour', 'Dans', 'Sur', 'Et', 'Comme', 'Mais', 'Ou', 'Où', 'Ne', 'Ni', 'Pas'];
        for (let i = 0; i < lowers.length; i++) {
            string = string.replace(new RegExp('\\s' + lowers[i] + '\\s', 'g'), function(txt) {
                return txt.toLowerCase();
            });
        }
    }

    const uppers = ['Id', 'R&d'];
    for (let i = 0; i < uppers.length; i++) {
        string = string.replace(new RegExp('\\b' + uppers[i] + '\\b', 'g'), uppers[i].toUpperCase());
    }

    return string;
}

console.log("the quick brown fox".titlecase('EN', true)); // "The quick brown fox"
console.log("le petit prince".titlecase('FR', true)); // "Le petit prince"

Solution 30: Locale-Aware Title Casing for Special Characters

This function is tailored for languages with special characters, using toLocaleLowerCase and toLocaleUpperCase for accurate transformations.

function toTitleCase(str) {
  return str.toLocaleLowerCase().replace(
    /(^|Ü|ü|Ş|ş|Ç|ç|İ|ı|Ö|ö|\w)\S*/g,
    (txt) => txt.charAt(0).toLocaleUpperCase() + txt.substring(1),
  )
}

console.log(toTitleCase('İSMAİL HAKKI')); // "İsmail Hakkı"
console.log(toTitleCase('ŞAHMARAN BİNBİR GECE MASALLARI')); // "Şahmaran Binbir Gece Masalları"

FAQs on Convert String to Title Case

Q: What is the simplest way to convert a string to title case in JavaScript

ANS: Using the split(), map(), and join() array methods is often considered the simplest and most readable approach for basic title casing: str.split(' ').map(w => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase()).join(' ').

Q: Can CSS alone handle title case conversion

ANS: CSS text-transform: capitalize; can capitalize the first letter of each word, but it has limitations with apostrophes, hyphens, and all-uppercase words, potentially leading to incorrect formatting.

Q: How can I preserve acronyms or specific capitalization in title case

ANS: You’ll need a more advanced JavaScript function that includes logic to identify and preserve acronyms or specific capitalization rules, often involving custom logic or regex patterns to avoid modifying those words. Solution 5 and Solution 26 demonstrate such approaches.

Q: Are there performance differences between regex and non-regex methods

ANS: Performance can vary depending on the complexity of the string, the specific regex pattern, and the JavaScript engine. While regex can be very powerful, a simple for loop might sometimes offer better performance for very large strings, though modern engines are highly optimized. Benchmarking is recommended for critical applications.

Q: How do I handle strings with special characters or accents

ANS: For handling special characters and accents, especially in non-English languages, use Unicode-aware regex patterns (like \p{L}) or locale-specific methods such as toLocaleUpperCase() and toLocaleLowerCase(). Solution 20 and Solution 30 provide examples.
