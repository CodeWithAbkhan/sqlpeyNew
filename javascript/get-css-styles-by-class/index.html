“Accessing CSS Styles by Class Name in JavaScript”

When developing for the web, dynamically accessing and manipulating CSS styles is a common requirement. This often involves targeting elements by their class names and retrieving the associated style rules. However, the browser’s Document Object Model (DOM) and Cascading Style Sheets (CSS) Object Model (CSSOM) can present complexities, especially when dealing with multiple stylesheets, dynamically loaded styles, or intricate selector specificity.

This query delves into various JavaScript techniques to extract CSS style definitions for a given class. We will examine several approaches, each with its nuances and browser compatibility considerations, to provide a robust set of solutions.

Solutions for Retrieving CSS Styles by Class Name

Here are several effective methods for accessing CSS styles associated with a class name:

Solutions

Solution 1: Basic Style Sheet Iteration

This approach iterates through the first stylesheet in the document and returns the cssText for a matching selector. It’s a straightforward method but limited to a single stylesheet.

function getStyle(className) {
    var cssText = "";
    // Accesses the first stylesheet and its rules
    var classes = document.styleSheets[0].rules || document.styleSheets[0].cssRules;
    for (var x = 0; x < classes.length; x++) {        
        // Checks if the selectorText matches the provided className
        if (classes[x].selectorText == className) {
            // Appends the CSS text or style text of the matching rule
            cssText += classes[x].cssText || classes[x].style.cssText;
        }         
    }
    return cssText;
}

Solution 2: Handling Multiple Style Sheets

This enhanced version iterates through all stylesheets present in the document, making it more versatile. It also ensures that if the cssText doesn’t contain the selector, it’s prepended.

/**
 * Retrieves styles associated with a class name across all stylesheets.
 * @param {string} className_ - The CSS class selector (e.g., '.my-class').
 * @returns {string|undefined} The CSS text for the matching selector, or undefined if not found.
 */
function getStyle(className_) {
    var styleSheets = window.document.styleSheets;
    var styleSheetsLength = styleSheets.length;

    // Loop through each stylesheet
    for(var i = 0; i < styleSheetsLength; i++){
        var classes = styleSheets[i].rules || styleSheets[i].cssRules;
        if (!classes) continue; // Skip if no rules are found

        var classesLength = classes.length;
        // Loop through each rule within the stylesheet
        for (var x = 0; x < classesLength; x++) {
            // Check if the selectorText matches the provided class name
            if (classes[x].selectorText === className_) {
                var ret;
                if (classes[x].cssText) {
                    ret = classes[x].cssText;
                } else {
                    ret = classes[x].style.cssText;
                }
                // Ensure the selector text is part of the returned CSS if not already present
                if (ret.indexOf(classes[x].selectorText) === -1) {
                    ret = classes[x].selectorText + "{" + ret + "}";
                }
                return ret; // Return the found CSS text
            }
        }
    }
    // Return undefined if no matching selector is found
}

Solution 3: Cross-Browser Property Retrieval

This solution offers two functions: one to get a specific CSS property value for an element identified by its class or ID, and another to retrieve the entire CSS text for a class. It leverages window.getComputedStyle for cross-browser compatibility.

/**
 * Retrieves a specific CSS property value for an element.
 * @param {string} classOrId - The selector for the element (e.g., '.my-class' or '#my-id').
 * @param {string} property - The CSS property name (e.g., 'position').
 * @returns {string} The computed value of the CSS property.
 */
function GetProperty(classOrId, property) { 
    var FirstChar = classOrId.charAt(0);
    var Remaining = classOrId.substring(1);
    // Select the element by ID or class name
    var elem = (FirstChar === '#') ? document.getElementById(Remaining) : document.getElementsByClassName(Remaining)[0];
    // Return the computed style property
    return window.getComputedStyle(elem, null).getPropertyValue(property);
}

/**
 * Retrieves the complete CSS text for a given class name.
 * @param {string} CLASSname - The CSS class selector (e.g., '.my-class').
 * @returns {string|undefined} The CSS text or undefined if not found.

function GetStyle(CLASSname) {
    var styleSheets = document.styleSheets;
    var styleSheetsLength = styleSheets.length;
    for (var i = 0; i < styleSheetsLength; i++) {
        var classes = styleSheets[i].rules || styleSheets[i].cssRules;
        if (!classes) { // Handle cases where cssRules might be null
            try {
                if (!styleSheets[i].cssRules) continue; // SecurityError in Firefox might cause this
            } catch (e) {

                    console.log("SecurityError: Cannot access stylesheet: " + styleSheets[i].href);
                    continue;
                }
            }
            classes = styleSheets[i].cssRules;

        for (var x = 0; x < classes.length; x++) {
            if (classes[x].selectorText === CLASSname) {
                var ret = (classes[x].cssText) ? classes[x].cssText : classes[x].style.cssText;
                // Ensure selector text is included if not present
                if (ret.indexOf(classes[x].selectorText) === -1) {
                    ret = classes[x].selectorText + "{" + ret + "}";
                }
                return ret;
            }
        }
    }
    return undefined; // Return undefined if no match is found
}

Solution 4: Robust Selector Matching with Normalization

This advanced solution handles more complex selectors, including combinators and multiple class names within a selector. It normalizes spacing and uses helper functions for splitting and checking for selector text presence.

// Encapsulated within a closure to avoid polluting the global scope
const getCssClasses = (function () {
    // Normalizes spacing around combinators and trims whitespace
    function normalize(str) {
        if (!str) return '';
        str = String(str).replace(/\s*([>~+])\s*/g, ' $1 '); // Normalize symbol spacing
        return str.replace(/(\s+)/g, ' ').trim(); // Normalize whitespace
    }

    // Splits a string, trims each part, and filters out empty strings
    function split(str, on) {
        return str.split(on).map(x => x.trim()).filter(x => x);
    }

    // Checks if a selector text contains any of the provided substrings
    function containsAny(selText, ors) {
        return selText ? ors.some(x => selText.indexOf(x) >= 0) : false;
    }

    return function (selector) {
        const logicalORs = split(normalize(selector), ','); // Split by comma for multiple selectors
        const sheets = Array.from(window.document.styleSheets);
        // Get all rules from all stylesheets
        const ruleArrays = sheets.map((x) => Array.from(x.rules || x.cssRules || []));
        const allRules = ruleArrays.reduce((all, x) => all.concat(x), []);
        
        // Filter rules that contain any of the normalized selectors
        return allRules.filter((x) => containsAny(normalize(x.selectorText), logicalORs));
    };
})();

Solution 6: Accessing Styles by Specific Stylesheet Item

This method targets a specific stylesheet using its item name (e.g., an ID assigned to the <style> tag) and then retrieves the CSS text for a particular class. This is useful if you have named style blocks.

/**
 * Retrieves CSS text for a class from a specifically identified stylesheet.
 * @param {string} className - The CSS class selector (e.g., '.test').
 * @returns {string|undefined} The CSS text of the class, or undefined if not found.
 */
function getStyle(className) {
    // Assumes 'menu' is the name/ID of the stylesheet element

Solution 7: Iterating Through All CSS Rules

This functional approach iterates through all stylesheets and all CSS rules within them. It provides a callback function that receives the selector, property name, value, and priority for each CSS declaration.

/**
 * Iterates through all CSS rules in all stylesheets, invoking a callback for each declaration.
 * @param {function} f - The callback function to execute for each CSS declaration.
 *   It receives (selector, propertyName, propertyValue, propertyPriority, cssRule).
 *   Returning false from the callback will stop iteration.
 */
function iterateCSS(f) {
  for (const styleSheet of window.document.styleSheets) {
    const classes = styleSheet.rules || styleSheet.cssRules;
    if (!classes) continue;

    for (const cssRule of classes) {
      // Ensure it's a style rule with a style object
      if (cssRule.type !== 1 || !cssRule.style) continue;
      
      const selector = cssRule.selectorText;
      const style = cssRule.style;
      if (!selector || !style.cssText) continue;

      // Iterate through each property in the style object
      for (let i = 0; i < style.length; i++) {
        const propertyName = style.item(i);
        // Execute the callback with details of the CSS declaration
        if (f(selector, propertyName, style.getPropertyValue(propertyName), style.getPropertyPriority(propertyName), cssRule) === false) {
          return; // Stop if callback returns false
        }
      }
    }
  }
}

Solution 9: Matching Class Names within Complex Selectors

This solution is designed to find styles where a given class is part of a more complex selector (e.g., .parent .child). It returns an array of objects, each containing the selectorText and its styleDefinition. It also includes a second function to gather all CSS definitions for elements within a specified subtree, using jQuery.

/**
 * Finds all CSS style blocks that involve a specific CSS selector.
 * @param {string} cssSelector - The CSS selector to search for (e.g., '.myClass' or '#myId').
 * @returns {Array<{selectorText: string, styleDefinition: string}>|null} An array of matching styles or null if none found.
 */
function getStyleWithCSSSelector(cssSelector) {
    var styleSheets = window.document.styleSheets;
    var styleSheetsLength = styleSheets.length;
    var arStylesWithCSSSelector = [];

    // Characters that typically follow a selector, used to avoid partial matches
    var arValidCharsAfterCssSelector = [" ", ".", ",", "#", ">", "+", ":", "["];

    // Iterate through all stylesheets
    for (var i = 0; i < styleSheetsLength; i++) {
        var classes = styleSheets[i].rules || styleSheets[i].cssRules;
        var classesLength = classes.length;
        for (var x = 0; x < classesLength; x++) {
            if (typeof classes[x].selectorText !== "undefined") {
                var matchClass = false;
                // Exact match
                if (classes[x].selectorText === cssSelector) {
                    matchClass = true;
                } else { // Check if the selector contains the target class as part of a larger selector
                    for (var j = 0; j < arValidCharsAfterCssSelector.length; j++) {
                        var cssSelectorWithNextChar = cssSelector + arValidCharsAfterCssSelector[j];
                        if (classes[x].selectorText.indexOf(cssSelectorWithNextChar) !== -1) {
                            matchClass = true;
                            break; // Exit inner loop once a match is found
                        }
                    }
                }

                if (matchClass === true) {
                    var styleDefinition;
                    if (classes[x].cssText) {
                        styleDefinition = classes[x].cssText;
                    } else {
                        styleDefinition = classes[x].style.cssText;
                    }
                    // Ensure selector text is included in the definition if not already
                    if (styleDefinition.indexOf(classes[x].selectorText) === -1) {
                        styleDefinition = classes[x].selectorText + "{" + styleDefinition + "}";
                    }
                    arStylesWithCSSSelector.push({ "selectorText": classes[x].selectorText, "styleDefinition": styleDefinition });
                }
            }
        }
    }
    return arStylesWithCSSSelector.length === 0 ? null : arStylesWithCSSSelector;
}

/**
 * Collects all CSS style definitions for elements within a specified DOM subtree.
 * Requires jQuery.
 * @param {string} selectorOfRootElement - A jQuery selector for the root element of the subtree.
 * @returns {string} A string containing all relevant CSS definitions.
 */
function getAllCSSClassDefinitionsForSubtree(selectorOfRootElement) {
    var arStackElements = []; // Stack for DFS traversal
    var existingClassDefinitions = {}; // Tracks added class definitions to avoid duplicates
    var rootElement = $(selectorOfRootElement)[0]; // Get the root DOM element
    var cssString = ""; // String to accumulate CSS output

    console.log(`Fetching all classes used in sub tree of ${selectorOfRootElement}`);
    arStackElements.push(rootElement);
    var currentElement;

    // Perform a Depth-First Search (DFS) traversal of the subtree
    while (currentElement = arStackElements.pop()) {
        currentElement = $(currentElement);
        console.log(`Processing element ${currentElement.attr("id")}`);

        var classesString = currentElement.attr("class");
        if (typeof classesString !== 'undefined') {
            var arClasses = classesString.split(" ");

            // Process each class associated with the current element
            for (var i = 0; i < arClasses.length; i++) {
                var arStylesWithCSSSelector = getStyleWithCSSSelector("." + arClasses[i]);
                console.log(`Processing class ${arClasses[i]}`);

                if (arStylesWithCSSSelector !== null) {
                    for (var j = 0; j < arStylesWithCSSSelector.length; j++) {
                        var tupleStyleWithCSSSelector = arStylesWithCSSSelector[j];
                        // Add style definition if it hasn't been added already
                        if (typeof existingClassDefinitions[tupleStyleWithCSSSelector.selectorText] === "undefined") {
                            cssString += tupleStyleWithCSSSelector.styleDefinition;
                            existingClassDefinitions[tupleStyleWithCSSSelector.selectorText] = true;
                        }
                    }
                }
            }
        }
        // Add child elements to the stack for processing
        if (currentElement.children().length > 0) {
            arStackElements = arStackElements.concat(currentElement.children().toArray());
        }
    }

    console.log(`Found ${Object.keys(existingClassDefinitions).length} CSS class definitions`);
    return cssString;
}

Solution 13: Parsing Styles into an Object

This function retrieves the CSS text for a class and then parses it into a JavaScript object where keys are style property names and values are their corresponding styles.

/**
 * Retrieves and parses CSS styles for a class name into an object.
 * @param {string} className - The CSS class selector (e.g., '.my-class').
 * @returns {object|false} An object of style properties and values, or false if not found.
 */
var getClassStyle = function(className) {
    var x, sheets, classes;
    for (sheets = document.styleSheets.length - 1; sheets >= 0; sheets--) {
        classes = document.styleSheets[sheets].rules || document.styleSheets[sheets].cssRules;
        for (x = 0; x < classes.length; x++) {
            if (classes[x].selectorText === className) {
                // Extract the style declarations within curly braces
                var classStyleTxt = (classes[x].cssText ? classes[x].cssText : classes[x].style.cssText).match(/\{\s*([^{}]+)\s*\}/)[1];
                var classStyles = {};
                // Match individual style declarations (property: value)
                var styleSets = classStyleTxt.match(/([^;:]+:\s*[^;:]+\s*)/g);
                for (y = 0; y < styleSets.length; y++) {
                    var style = styleSets[y].match(/\s*([^:;]+):\s*([^;:]+)/);
                    if (style && style.length > 2)
                        classStyles[style[1]] = style[2]; // Store as key-value pair
                }
                return classStyles;
            }
        }
    }
    return false;
};

Solution 14: Key/Value Object of Styles by Selector

This function searches all stylesheets for matching selectors and returns an object where keys are selectors and values are their corresponding style content. It includes an option to match selectors starting with a given string.

/**
 * Retrieves CSS style blocks matching a selector from all stylesheets.
 * @param {string} className - The CSS selector to match.
 * @param {boolean} [startingWith=false] - If true, matches selectors that start with `className`.
 * @returns {object|null} An object with matching selectors as keys and style content as values, or null if no matches.
 * @throws {Error} If `className` is not provided.
 */
function getStylesBySelector(className, startingWith) {
    if (!className || className === '') throw new Error('Please provide a css class name');

    var styleSheets = window.document.styleSheets;
    var result = {};

    // Iterate through all stylesheets
    for (var i = 0, l = styleSheets.length; i < l; i++) {
        var classes = styleSheets[i].rules || styleSheets[i].cssRules || [];

        // Iterate through all rules in the current stylesheet
        for (var x = 0, ll = classes.length; x < ll; x++) {
            var selector = classes[x].selectorText || '';
            var content = classes[x].cssText || classes[x].style.cssText || '';

            // Check if the selector matches (either exact or startsWith)
            if ((startingWith && selector.indexOf(className) === 0) || selector === className) {
                // Extract style content from within curly braces
                var styleContent = content.split(/(?:{|})/)[1]?.trim(); 
                if (styleContent) {
                    result[selector] = styleContent;
                }
            }
        }
    }

    // Return the result object if it contains matches, otherwise null
    return Object.keys(result).length > 0 ? result : null;
}

Solution 15: Applying Styles as Inline Styles and Removing Classes

This solution provides two functions. getStyle retrieves CSS text for a class, potentially including it as part of a selector path. setInlineStyle then applies these retrieved styles as inline styles to a given DOM node and removes the node’s classes.

/**
 * Retrieves CSS text for a class, handling cases where the class is part of a selector path.
 * @param {string} cssClass - The CSS class selector (e.g., '.my-class').
 * @returns {string} A string of concatenated CSS declarations.
 */
function getStyle(cssClass) {
    let result = '';
    let styleSheets = document.styleSheets;
    for (let i = 0, j = styleSheets.length; i < j; i++) {
        let classes = styleSheets[i].cssRules;
        if (classes && classes.length) {
            for (let x = 0, y = classes.length; x < y; x++) {
                let selectorText = classes[x].selectorText;
                if (selectorText && selectorText.includes(cssClass)) {
                    let rules = selectorText.split(',');
                    for (let n = 0, m = rules.length; n < m; n++) {
                        let rule = rules[n];
                        if (rule) {
                            let rulePath = rule.trim().split(' ');
                            let p = rulePath[rulePath.length - 1]; // Get the last part of the selector
                            if (p === cssClass) { // Check if it matches the target class
                                let ret = classes[x].cssText || classes[x].style.cssText || '';
                                let css = ret.match(/{[\w\W]*}/); // Extract styles within braces
                                let cssText = css && css[0] ? css[0].substring(1, css[0].length - 1).trim() : '';
                                result += cssText;
                            }
                        }
                    }
                }
            }
        }
    }
    return result;
}

Solution 16: Filtering Styles into an Object with Lodash

This solution, leveraging Lodash’s _.pickBy, retrieves styles for a given class name and filters them into an object containing only valid, non-initial string properties.

/**
 * Retrieves and filters CSS style properties for a class name.
 * @param {string} className_ - The CSS class selector (e.g., '.my-class').
 * @returns {object|undefined} An object of filtered style properties, or undefined if not found.
 */
function getStyle(className_) {
    var styleSheets = window.document.styleSheets;
    var styleSheetsLength = styleSheets.length;
    for (var i = 0; i < styleSheetsLength; i++) {
        var classes = styleSheets[i].rules || styleSheets[i].cssRules;
        if (!classes) continue;
        var classesLength = classes.length;
        for (var x = 0; x < classesLength; x++) {
            if (classes[x].selectorText === className_) {
                // Use _.pickBy to filter properties:
                // - k is not a number (numeric indices are skipped)
                // - v is a string
                // - v is not empty
                // - v is not 'initial'
                // - k is not 'cssText'
                return _.pickBy(classes[x].style, (v, k) => isNaN(parseInt(k)) && typeof(v) === 'string' && v && v !== 'initial' && k !== 'cssText');
            }
        }
    }
    return undefined; // Return undefined if no match is found
}
