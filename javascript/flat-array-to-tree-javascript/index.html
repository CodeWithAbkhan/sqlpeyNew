“Efficiently Convert Flat Arrays to Tree Structures in JavaScript”

When dealing with data that has parent-child relationships represented in a flat array, converting it into a nested tree structure is a common requirement. This is often seen in scenarios like displaying nested comments, file system directories, or organizational charts. Several JavaScript approaches can achieve this, each with its own advantages regarding performance, code conciseness, and dependencies.

This post examines various effective methods to build a tree from a flat array, offering practical code examples and discussing their nuances.

Solution 1: Map-Lookup for Efficient Tree Construction

This solution leverages a map (JavaScript object) to store references to each node, allowing for efficient lookups. It’s particularly fast when parent nodes consistently appear before their children, but can be adapted for unsorted data. This method handles multiple roots and can be modified to ignore or report dangling branches. It requires no third-party libraries and is highly performant, with a time complexity of O(n log n) or O(n) depending on the map implementation.

function listToTree(list) {
  const map = {};
  const roots = [];
  
  // First pass: create a map and initialize children arrays
  for (let i = 0; i < list.length; i++) {
    map[list[i].id] = i; // Store index for quick access
    list[i].children = []; // Initialize children property
  }
  
  // Second pass: build the tree structure
  for (let i = 0; i < list.length; i++) {
    const node = list[i];
    if (node.parentId !== "0") { // Assuming "0" indicates a root parent
      // Check if parent exists to handle potential dangling branches
      if (map[node.parentId] !== undefined) {
        list[map[node.parentId]].children.push(node);
      } else {
        // Handle dangling branch if necessary, e.g., skip or log
      }
    } else {
      roots.push(node); // Add root nodes to the roots array
    }
  }
  return roots;
}

Solution 2: Hash Table for Unordered Data

This efficient plain JavaScript solution uses a hash table to build the tree, even if the input array is not sorted. It iterates through the data once to create the hash table and then again to link nodes. This approach is praised for its performance without external libraries.

const createDataTree = (dataset) => {
  const hashTable = Object.create(null); // Use Object.create for a clean hash table

  // First pass: Populate hash table and initialize children arrays
  dataset.forEach(aData => {
    hashTable[aData.ID] = { ...aData, children: [] };
  });

  const dataTree = [];
  // Second pass: Link nodes
  dataset.forEach(aData => {
    if (aData.parentID) { // If it has a parent
      // Ensure the parent exists in the hashTable before pushing
      if (hashTable[aData.parentID]) {
        hashTable[aData.parentID].children.push(hashTable[aData.ID]);
      }
    } else { // If it's a root node
      dataTree.push(hashTable[aData.ID]);
    }
  });
  return dataTree;
};

Solution 3: Recursive Filtering with Underscore.js

This method, while effective, relies on the underscore.js library for its filter and each functions. It uses recursion to traverse the array and build the tree. It assumes the presence of elements with a parent ID of 0 for root nodes. Orphaned elements and their descendants will not be included in the resulting tree.

// Ensure underscore.js is loaded: <script src="https://cdnjs.cloudflare.com/ajax/libs/underscore.js/1.9.1/underscore-min.js"></script>

let unflatten = function(array, parent, tree) {
    tree = typeof tree !== 'undefined' ? tree : [];
    parent = typeof parent !== 'undefined' ? parent : { id: 0 }; // Default to root parent

    // Find direct children of the current parent
    const children = _.filter(array, function(child) {
        return child.parentid === parent.id;
    });

        if (parent.id === 0) { // If it's a root level
            tree = children;
        } else {
            parent['children'] = children; // Attach children to the parent
        }
        // Recursively process children
        _.each(children, function(child) {
            unflatten(array, child);
        });
    }

    return tree;
};

Solution 5: Vanilla JS with Hash Table (Unordered Data)

This vanilla JavaScript solution is robust for unordered data. It first builds a hash table (object) for quick access to each item by its ID, then iterates again to assign children to their respective parents.

const unflatten = (arr) => {
  const tree = [];
  const mappedArr = {};
  let arrElem, mappedElem;

  // First map: create a hash table of all elements
  for (let i = 0, len = arr.length; i < len; i++) {
    arrElem = arr[i];
    mappedArr[arrElem.id] = arrElem;
    mappedArr[arrElem.id]['children'] = []; // Initialize children array
  }

  // Second map: build the tree structure
  for (const id in mappedArr) {
    if (mappedArr.hasOwnProperty(id)) {
      mappedElem = mappedArr[id];
      // If element has a parent and the parent exists in the map
      if (mappedElem.parentid && mappedArr[mappedElem['parentid']]) {
        mappedArr[mappedElem['parentid']]['children'].push(mappedElem);
      } else {
        // Add to the root level if no parent or parent not found
        tree.push(mappedElem);
      }
    }
  }
  return tree;
};

Solution 11: Optimized Single-Pass with Hash Table (ES6+)

This advanced ES6 solution uses a single pass with a hash table (t) to build the tree structure efficiently. It handles unordered data and creates a more structured output, grouping by parent ID.

const getTree = (data, rootParentId) => {
    const hashTable = {}; // Use object as a hash table

    // First pass: Populate hash table and initialize children arrays
    data.forEach(item => {
        // Ensure parent and current item entries exist in hashTable
        hashTable[item.parentId] = hashTable[item.parentId] ?? { children: [] };
        hashTable[item.id] = hashTable[item.id] ?? {};
        
        // Add current item's data and found children reference
        Object.assign(hashTable[item.id], item);
        
        // Add item to its parent's children array
        hashTable[item.parentId].children.push(hashTable[item.id]);
    });

    // Return the children of the root parent
    return hashTable[rootParentId]?.children ?? [];
};

Solution 12: ES6 Tree Conversion with HTML Generation

This solution provides two ES6 functions: one to convert a flat array into a tree structure keyed by IDs, and another to recursively generate an HTML <ul><li> structure from the tree. It allows for custom callbacks during LI creation, adding interactivity.

/**
 * Convert nodes list related by parent ID to a tree.
 * @param {Array} arr Array of nodes
 * @param {string} p Property name for parent ID (defaults to "parent_id")
 * @returns {Object} Nodes tree keyed by ID
 */
const getTree = (arr, p = "parent_id") => arr.reduce((o, n) => {
  // Ensure node and potential parent entries exist in the accumulator
  if (!o[n.id]) o[n.id] = {};
  if (!o[n[p]]) o[n[p]] = {};
  if (!o[n[p]].nodes) o[n[p]].nodes = []; // Initialize nodes array for parent

  // If the current node already has children (from a previous pass), transfer them
  if (o[n.id].nodes) n.nodes = o[n.id].nodes;

  o[n[p]].nodes.push(n); // Add current node to its parent's children
  o[n.id] = n; // Update the node in the accumulator

  return o;
}, {}); // Initial accumulator is an empty object

/**
 * Convert Tree structure to UL>LI and append to an Element.
 * @param {Array} tree Tree array of nodes
 * @param {Element} el HTMLElement to insert into
 * @param {function} cb Callback function called on every LI creation
 */
const treeToHTML = (tree, el, cb) => el.append(tree.reduce((ul, n) => {
  const li = document.createElement('li');

  if (cb) cb.call(li, n); // Execute callback for LI customization
  if (n.nodes?.length) treeToHTML(n.nodes, li, cb); // Recursively process children

  ul.append(li); // Append the LI to the UL
  return ul;
}, document.createElement('ul'))); // Start with an empty UL

Solution 14: Mutating Array with Hash Table Cache

This approach uses a hash table to cache elements by ID. It then iterates through the data, adding children to their parents and removing processed items from the root array. This modifies the original array and requires careful index management due to splice.

function listToTree(data, options) {
  options = options || {};
  var ID_KEY = options.idKey || 'id';
  var PARENT_KEY = options.parentKey || 'parent';
  var CHILDREN_KEY = options.childrenKey || 'children';

  var map = {};
  // First pass: create a cache (map) of elements by ID and initialize children arrays
  for (var i = 0; i < data.length; i++) {
    if (data[i][ID_KEY]) {
      map[data[i][ID_KEY]] = data[i];
      data[i][CHILDREN_KEY] = [];
    }
  }

  // Second pass: build the tree structure
  for (var i = 0; i < data.length; i++) {
    if (data[i][PARENT_KEY]) { // If the item has a parent key
      if (map[data[i][PARENT_KEY]]) { // Check if the parent exists in the cache
        map[data[i][PARENT_KEY]][CHILDREN_KEY].push(data[i]); // Add item to parent's children
        data.splice(i, 1); // Remove the item from the root array
        i--; // Adjust iterator after splice
      } else {
        data[i][PARENT_KEY] = 0; // Handle potentially dirty data by resetting parentId
      }
    }
  }
  return data; // The remaining items in 'data' are the root elements
}

Solution 15: Generic TypeScript/JavaScript Function

This adaptable solution provides a generic function for converting flat lists to trees, usable in both JavaScript and TypeScript. It uses a hash map and a loop to organize children, with parameters for ID, parent ID, children list path, and a check for root nodes.

// Javascript Implementation:
export const flatListToTreeJS = (flatList, idPath, parentIdPath, childListPath, isParent) => {
  const rootParents = [];
  const map = {};
  
  // First pass: Build map and initialize children arrays
  for (const item of flatList) {
    if (!item[childListPath]) item[childListPath] = [];
    map[item[idPath]] = item;
  }

  // Second pass: Link nodes
  for (const item of flatList) {
    const parentId = item[parentIdPath];
    if (isParent(item)) { // Check if this item is a root node
      rootParents.push(item);
    } else {
      const parentItem = map[parentId];
      if (parentItem) { // Ensure parent exists before pushing
        parentItem[childListPath].push(item);
      }
    }
  }
  return rootParents;
};

  const map: any = {}; // Use 'any' or more specific type for map

    }
  }

};

    const hieArr = convertToHieArr(hieObj, "Top Level");

    function createHieobj(data, seq, ind) {
        const s = seq[ind];
        if (s === undefined) return [];

        const childObj = {};

function unflat(data, options = {}) {
    const { id, parentId, childrenKey } = {
        id: "id",
        parentId: "parentId",
        childrenKey: "children",
        ...options // Override defaults with provided options
    };

    // Create a map of items by their ID for efficient lookup
    const copiesById = data.reduce(
        (copies, datum) => ((copies[datum[id]] = datum) && copies),
        {}
    );

    // Reduce to build the tree structure
    return Object.values(copiesById).reduce(
        (root, datum) => {
            // Check if datum has a parentId and if the parent exists in the map
            if (datum[parentId] && copiesById[datum[parentId]]) {
                // Add datum to its parent's children array
                copiesById[datum[parentId]][childrenKey] = [
                    ...(copiesById[datum[parentId]][childrenKey] || []), // Ensure children array exists
                    datum
                ];
            } else {
                // If no parentId or parent not found, consider it a root node
                root = [...root, datum];
            }
            return root;
        }, [] // Initial value for the reducer is an empty root array
    );
}

Solution 21: Vanilla JS Bi-directional Tree Conversion

This efficient vanilla JavaScript function offers bi-directional mapping (root-to-leaves and leaves-to-root), returns all nodes, roots, and leaves, and performs the conversion in a single data pass.

/**
 * Converts a flat array of items into a tree structure.
 * @param {Array} data - The array of items.
 * @param {string} idKey - The key for the item's ID (e.g., 'id').
 * @param {string} parentIdKey - The key for the item's parent ID (e.g., 'parentId').
 * @param {*} [noParentValue=null] - The value indicating a root item's parentId (e.g., null, 0).
 * @param {boolean} [bidirectional=true] - Whether to add parent references.
 * @returns {{roots: Object, nodes: Object, leaves: Object}} An object containing roots, all nodes, and leaves.
 */
function flatToTree(data, idKey, parentIdKey, noParentValue = null, bidirectional = true) {
  const nodes = {}, roots = {}, leaves = {};

  // Iterate over all data items
  for (const item of data) {
    // Add item as a node, potentially as a leaf initially
    if (nodes[item[idKey]]) {
      // If node already exists (child found first), merge data and children
      nodes[item[idKey]] = Object.assign(nodes[item[idKey]], item);
    } else {
      // Add new node and assume it's a leaf for now
      nodes[item[idKey]] = Object.assign({ $children: [] }, item);
      leaves[item[idKey]] = nodes[item[idKey]];
    }

    // Process parent-child relationship
    if (item[parentIdKey] !== noParentValue) { // If item has a parent
      if (nodes[item[parentIdKey]]) { // If parent node already exists
        (nodes[item[parentIdKey]].$children || []).push(nodes[item[idKey]]); // Add as child
      } else { // If parent node does not exist yet
        // Create a placeholder for the parent and add the current item as its child
        nodes[item[parentIdKey]] = { $children: [nodes[item[idKey]]] };
      }
      if (bidirectional) {
        nodes[item[idKey]].$parent = nodes[item[parentIdKey]]; // Link to parent
      }
      delete leaves[item[idKey]]; // Item is not a leaf if it has a parent
    } else { // If item is a root node
      roots[item[idKey]] = nodes[item[idKey]];
    }
  }
  return { roots, nodes, leaves };
}

  var childrenArray = array.filter(function(child) {

    });
  }

function unflatten(records, rootCategoryId, parent, tree) {
    if (!_.isArray(tree)) { // Initialize tree if not already an array
        tree = [];
        _.each(records, function(rec) {
            // Check if the record's parentId contains the root identifier
            if (rec.parentId.indexOf(rootCategoryId) >= 0) {
                const tmp = _.cloneDeep(rec); // Deep clone to avoid reference issues
                // Find direct children of this root item
                tmp.children = _.filter(records, function(r) {
                    return r.parentId === tmp.id;
                });
                tree.push(tmp);
                // Recursively process children of this root item
                _.each(tmp.children, function(child) {
                    return unflatten(records, rootCategoryId, child, tree);
                });
            }
        });
    } else { // If tree is already initialized, process children of a given parent
        if (parent) {
            parent.children = _.filter(records, function(r) {
                return r.parentId === parent.id;
            });
            _.each(parent.children, function(child) {
                return unflatten(records, rootCategoryId, child, tree);
            });
        }
    }
    return tree;
}

function list2tree(data, opt) {
    opt = opt || {};
    const KEY_ID = opt.key_id || 'ID';
    const KEY_PARENT = opt.key_parent || 'FatherID';
    const KEY_CHILD = opt.key_child || 'children';
    const EMPTY_CHILDREN = opt.empty_children;
    const ROOT_ID = opt.root_id || 0;
    const MAP = opt.map || {};

    function getNode(id) {
        let node = [];
        for (let i = 0; i < data.length; i++) {
            if (data[i][KEY_PARENT] == id) {
                // Map custom properties if provided
                for (const k in MAP) {
                    data[i][k] = data[i][MAP[k]];
                }
                // Recursively get children
                const children = getNode(data[i][KEY_ID]);
                if (children !== undefined) {
                    data[i][KEY_CHILD] = children;
                } else {
                    // Handle empty children based on option
                    if (EMPTY_CHILDREN === null) data[i][KEY_CHILD] = null;
                    else if (JSON.stringify(EMPTY_CHILDREN) === '[]') data[i][KEY_CHILD] = [];
                }
                node.push(data[i]);
            }
        }
        return node.length === 0 ? undefined : node;
    }
    return getNode(ROOT_ID);
}
