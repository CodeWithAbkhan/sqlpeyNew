“JavaScript File MIME Type Detection: Client-Side Inspection Methods”

How can a web application accurately ascertain the true MIME type of a file selected by a user via a client-side input element, especially when relying solely on the File.type property is unreliable due to potential spoofing via renaming?

While server-side verification remains the preferred security measure, robust client-side checks offer valuable immediate feedback to the user. Modern browsers offer the File API, which, when combined with binary data reading, allows for inspecting file headers—the so-called “magic numbers”—to verify authenticity.

Retrieving File Information via Input or Drag-and-Drop

The initial step involves accessing the file object(s) from either an <input type="file"> element or a drag-and-drop zone. Browser support for FileReader and Blob should be confirmed first.

if (window.FileReader && window.Blob) {
    // File and Blob APIs are available.
} else {
    // Fallback needed: File and Blob are not supported in this environment.
}

Using a Standard File Input

When a file selection changes, the files property of the input control holds the FileList.

const fileControl = document.getElementById("fileInputSelector"); // Assume this ID is used
fileControl.addEventListener("change", function(event) {
    const files = fileControl.files;
    for (let i = 0; i < files.length; i++) {
        console.log(`Filename: ${files[i].name}`);
        console.log(`Browser Reported Type: ${files[i].type}`);
        console.log(`Size: ${files[i].size} bytes`);
    }
}, false);

Implementing Drag and Drop File Handling

For drag-and-drop interfaces, the files are accessed via event.dataTransfer.files.

const dropTarget = document.getElementById("dropZoneArea"); // Assume this ID is used
dropTarget.addEventListener("dragover", function(event) {
    event.preventDefault();
}, false);

dropTarget.addEventListener("drop", function(event) {
    event.preventDefault();

    for (let i = 0; i < files.length; i++) {

        console.log(`Browser Reported Type: ${files[i].type}`);
        console.log(`Size: ${files[i].size} bytes`);
    }
}, false);

Methods for MIME Type Assessment

Two primary techniques exist for client-side MIME type derivation:

1. Naïve Property Check (Unreliable)

The simplest approach is relying on blob.type or file.type.

const suspiciousBlob = files[i]; // Referencing a File/Blob object from Step 1
const reportedMime = suspiciousBlob.type;
console.log(`Reported MIME: ${reportedMime}`); 
// Examples: 'image/jpeg', 'image/png'

Caveat: This value is derived from the file extension (e.g., .jpg, .png) provided by the operating system or browser and can be trivially bypassed by renaming a file (e.g., renaming malicious.exe to safe.png).

2. Robust Header Inspection (Magic Numbers)

To confirm the actual content type, the initial bytes of the file must be read and compared against known file signatures, or “magic numbers.” This requires using FileReader to read a small portion of the file as an ArrayBuffer.

Reading only the beginning bytes avoids memory exhaustion associated with loading large files entirely.

Extracting the Header

The core operation reads the initial few bytes:

const fileBlob = files[i]; // The File/Blob object
const fileReader = new FileReader();

fileReader.onloadend = function(e) {
  // Extract the first 4 bytes and convert to hexadecimal string
  const arr = (new Uint8Array(e.target.result)).subarray(0, 4);
  let headerHex = "";
  for(let j = 0; j < arr.length; j++) {
     headerHex += arr[j].toString(16).padStart(2, '0'); // Ensure two digits per byte
  }
  console.log(`File Header (Hex): ${headerHex}`);

  // Determine the real MIME type based on the header
  const realType = determineMimeFromHeader(headerHex);
  console.log(`Validated MIME Type: ${realType}`);
};

fileReader.readAsArrayBuffer(fileBlob);

Mapping Header to MIME Type

A switch statement or lookup table maps the resulting hex string to the correct MIME type. For JPEG, multiple signatures exist due to format evolution:

function determineMimeFromHeader(header) {
    switch (header) {
        case "89504e47": // PNG Signature: 89 50 4E 47
            return "image/png";
        case "47494638": // GIF Signature: 47 49 46 38 (GIF87a or GIF89a)
            return "image/gif";
        case "ffd8ffe0":
        case "ffd8ffe1":
        case "ffd8ffe2":
        case "ffd8ffe3":
        case "ffd8ffe8": // JPEG Signatures (starts with FF D8 FF)
            return "image/jpeg";
        default:
            // Fallback to the browser's suggestion or 'unknown'
            return "unknown"; 
    }
}

Alternative: Efficient Byte Comparison with Masking

An alternative, more generalized solution involves defining patterns and masks for comparison, which prevents reading the entire file content and handles various signature lengths efficiently, as demonstrated by Victor’s approach:

/**
 * Efficiently load MIME type based on signature from initial bytes.
 * Only reads a minimal slice (e.g., first 4 bytes).
 * @param  {File}   file        The File instance.
 * @param  {Function} callback  Function to receive the result.
 */
function checkSignatureMime(file, callback) {
    const recognizedMimes = [
        {
            mime: 'image/jpeg',
            pattern: [0xFF, 0xD8, 0xFF], // Start of Image marker + JFIF identifier start
            mask: [0xFF, 0xFF, 0xFF],
        },
        {
            mime: 'image/png',
            pattern: [0x89, 0x50, 0x4E, 0x47], // PNG signature
            mask: [0xFF, 0xFF, 0xFF, 0xFF],
        }
        // Reference: https://mimesniff.spec.whatwg.org/#matching-an-image-type-pattern
    ];

    function performCheck(bytes, mimeDef) {
        for (let i = 0, l = mimeDef.mask.length; i < l; ++i) {
            if ((bytes[i] & mimeDef.mask[i]) !== mimeDef.pattern[i]) {
                return false;
            }
        }
        return true;
    }

    const minimalSlice = file.slice(0, 4); // Read only the first 4 bytes

    const reader = new FileReader();
    reader.onloadend = function(e) {
        if (e.target.readyState === FileReader.DONE) {
            const bytes = new Uint8Array(e.target.result);

            for (const mimeDef of recognizedMimes) {
                if (performCheck(bytes, mimeDef)) {
                    return callback(`Mime: ${mimeDef.mime} | Browser: ${file.type}`);
                }
            }
            return callback(`Mime: unknown | Browser: ${file.type}`);
        }
    };
    reader.readAsArrayBuffer(minimalSlice);
}

Alternative 3: Utilizing Third-Party Libraries

For extensive format support without maintaining numerous magic number lists, specialized libraries are beneficial. The file-type package is designed for this and can operate in the browser by reading only the minimum necessary bytes:

import fileType from 'file-type'; 

const requiredBytes = fileType.minimumBytes; 
const slice = file.slice(0, requiredBytes);

const fileReader = new FileReader();
fileReader.onloadend = function(e) {
  if (e.target.readyState !== FileReader.DONE) return;

  const uint8Bytes = new Uint8Array(e.target.result);
  const result = fileType.fromBuffer(uint8Bytes);

  // result contains { ext, mime }
  console.log(`Detected Mime: ${result.mime}`);
};
fileReader.readAsArrayBuffer(slice);

Alternative 4: Image Loading Check (For Images Only)

If you only deal with images, attempting to load the file’s Data URL into an <img> element and observing the onerror event is a practical heuristic:

function checkImageValidity(fileDataUrl, callback) {
    const tempImage = new Image();
    tempImage.onload = function() { callback(true); }; // Load succeeded
    tempImage.onerror = function() { callback(false); }; // Failed to render (wrong format)
    tempImage.src = fileDataUrl;
}

// Usage when using FileReader.readAsDataURL(file):
reader.onload = function (e) {
    checkImageValidity(e.target.result, (isValid) => {
        if (isValid) {
            // Process as valid image
        } else {
            // Revert file input or show error
        }
    });
};

Alternative 5: Resetting the Input Field

When a file fails client-side validation (especially using header checks), resetting the form element can be necessary to allow the user to select a new file for the same input field:

// Example where validation fails and the form is reset
if (!isHeaderValid) {
    alert("Invalid file format detected.");
    document.getElementById("form-id")[0].reset(); // Resets the specific file input element
    return;
}

Further Depth on PNG Validation

For the Portable Network Graphics (PNG) format specifically, validation can be extended beyond the initial 8-byte signature (89504E47) by checking the sequence and integrity (CRC checksums) of subsequent data chunks, as defined in the PNG specification. Libraries like png-validator implement this deeper level of structural verification.

FAQs on JavaScript File MIME Type Detection

Q: Why is relying on file.type unreliable for validation

ANS: The file.type property usually reflects the file extension provided by the OS or browser, which can be altered by renaming the file to match an accepted extension, making it insecure for critical validation.

Q: What are magic numbers in file validation

ANS: Magic numbers are specific sequences of bytes located at the beginning of a file that uniquely identify its format, allowing for verification independent of the file extension.

Q: Should server-side validation be skipped if client-side checks are thorough

ANS: No, server-side validation is essential. Client-side checks are primarily for user experience and immediate feedback; they can always be bypassed by sophisticated users or automated scripts.

Q: How can I check for a PDF file signature in JavaScript

ANS: A common PDF signature is the string %PDF (hex 25504446). You can read the first few bytes using FileReader and convert them to a string to check if it matches this pattern.

function getBLOBFileHeader(url, blob, callback) {
  var fileReader = new FileReader();
  fileReader.onloadend = function(e) {
    var arr = (new Uint8Array(e.target.result)).subarray(0, 4);

    for (var i = 0; i &lt; arr.length; i++) {
      header += arr[i].toString(16);
    }
    callback(url, header);
  };
  fileReader.readAsArrayBuffer(blob);
}

/**
 * Load the mime type based on the signature of the first bytes of the file
 * @param  {File}   file        A instance of File
 * @param  {Function} callback  Callback with the result
 * @author Victor www.vitim.us
 * @date   2017-03-23
 */
function loadMime(file, callback) {
    
    //List of known mimes
    var mimes = [
        {
            mime: 'image/jpeg',
            pattern: [0xFF, 0xD8, 0xFF],
            mask: [0xFF, 0xFF, 0xFF],
        },
        {
            mime: 'image/png',
            pattern: [0x89, 0x50, 0x4E, 0x47],
            mask: [0xFF, 0xFF, 0xFF, 0xFF],
        }
        // you can expand this list @see https://mimesniff.spec.whatwg.org/#matching-an-image-type-pattern
    ];

    function check(bytes, mime) {
        for (var i = 0, l = mime.mask.length; i &lt; l; ++i) {
            if ((bytes[i] &amp; mime.mask[i]) - mime.pattern[i] !== 0) {
                return false;
            }
        }
        return true;
    }

    var blob = file.slice(0, 4); //read the first 4 bytes of the file

    var reader = new FileReader();
    reader.onloadend = function(e) {
        if (e.target.readyState === FileReader.DONE) {
            var bytes = new Uint8Array(e.target.result);

            for (var i=0, l = mimes.length; i&lt;l; ++i) {
                if (check(bytes, mimes[i])) return callback("Mime: " + mimes[i].mime + " &lt;br&gt; Browser:" + file.type);
            }

            return callback("Mime: unknown &lt;br&gt; Browser:" + file.type);
        }
    };
    reader.readAsArrayBuffer(blob);
}
