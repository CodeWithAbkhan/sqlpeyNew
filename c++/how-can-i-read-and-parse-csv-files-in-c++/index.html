Top C++ Solutions for Parsing CSV Files

Parsing Comma Separated Value (CSV) files in C++ can range from a straightforward task to a complex challenge, depending on the adherence of the CSV to standard specifications. Factors like embedded commas, quoted fields, escaped quotes, and multi-line fields can complicate simple parsing logic. This comprehensive guide explores various C++ approaches, from fundamental methods to powerful libraries, to help you efficiently and reliably parse CSV data.

Solution 1: Basic Line-by-Line Parsing with std::getline and std::stringstream

This approach focuses on reading each line and then using std::stringstream to split the line based on the comma delimiter. It’s a good starting point for simpler CSV files.

#include <vector>
#include <string>
#include <sstream>
#include <iostream>

std::vector<std::string> getNextLineAndSplitIntoTokens(std::istream& str) {
    std::vector<std::string> result;
    std::string line;
    std::getline(str, line);

    std::stringstream lineStream(line);
    std::string cell;

    while (std::getline(lineStream, cell, ',')) {
        result.push_back(cell);
    }
    // This checks for a trailing comma with no data after it.
    if (!lineStream && cell.empty()) {
        // If there was a trailing comma then add an empty element.
        result.push_back("");
    }
    return result;
}

// Example of a class to represent a CSV row for more structured data handling
class CSVRow {
public:
    std::string_view operator[](std::size_t index) const {
        return std::string_view(&m_line[m_data[index] + 1], m_data[index + 1] - (m_data[index] + 1));
    }

    std::size_t size() const {
        return m_data.size() - 1;
    }

    void readNextRow(std::istream& str) {
        std::getline(str, m_line);

        m_data.clear();
        m_data.emplace_back(-1);
        std::string::size_type pos = 0;
        while ((pos = m_line.find(',', pos)) != std::string::npos) {
            m_data.emplace_back(pos);
            ++pos;
        }
        // This checks for a trailing comma with no data after it.
        pos = m_line.size();
        m_data.emplace_back(pos);
    }

private:
    std::string m_line;
    std::vector<int> m_data;
};

std::istream& operator>>(std::istream& str, CSVRow& data) {
    data.readNextRow(str);
    return str;
}

// Using an iterator for a more C++ idiomatic approach
class CSVIterator {
public:
    typedef std::input_iterator_tag iterator_category;
    typedef CSVRow value_type;
    typedef std::size_t difference_type;
    typedef CSVRow* pointer;
    typedef CSVRow& reference;

    CSVIterator(std::istream& str) : m_str(str.good() ? &str : nullptr) { ++(*this); }
    CSVIterator() : m_str(nullptr) {}

    // Pre Increment
    CSVIterator& operator++() {
        if (m_str) {
            if (!((*m_str) >> m_row)) {
                m_str = nullptr;
            }
        }
        return *this;
    }
    // Post increment
    CSVIterator operator++(int) {
        CSVIterator tmp(*this);
        ++(*this);
        return tmp;
    }
    CSVRow const& operator*() const { return m_row; }
    CSVRow const* operator->() const { return &m_row; }

    bool operator==(CSVIterator const& rhs) { return ((this == &rhs) || ((this->m_str == nullptr) && (rhs.m_str == nullptr))); }
    bool operator!=(CSVIterator const& rhs) { return !((*this) == rhs); }

private:
    std::istream* m_str;
    CSVRow m_row;
};

class CSVRange {

public:

Solution 2: State Machine for Excel-Compatible CSV Parsing

This method employs a finite-state machine to handle the nuances of CSV parsing, particularly quoting and escaped quotes, as often seen in Microsoft Excel exports.

#include <istream>
#include <string>
#include <vector>

enum class CSVState {
    UnquotedField,
    QuotedField,
    QuotedQuote
};

std::vector<std::string> readCSVRow(const std::string& row) {
    CSVState state = CSVState::UnquotedField;
    std::vector<std::string> fields{""};
    size_t i = 0; // index of the current field
    for (char c : row) {
        switch (state) {
            case CSVState::UnquotedField:
                switch (c) {
                    case ',': // end of field
                        fields.push_back(""); i++;
                        break;
                    case '"': state = CSVState::QuotedField;
                        break;
                    default:  fields[i].push_back(c);
                        break;
                }
                break;
            case CSVState::QuotedField:
                switch (c) {
                    case '"': state = CSVState::QuotedQuote;
                        break;
                    default:  fields[i].push_back(c);
                        break;
                }
                break;
            case CSVState::QuotedQuote:
                switch (c) {
                    case ',': // , after closing quote
                        fields.push_back(""); i++;
                        state = CSVState::UnquotedField;
                        break;
                    case '"': // "" -> "
                        fields[i].push_back('"');
                        state = CSVState::QuotedField;
                        break;
                    default:  // end of quote
                        state = CSVState::UnquotedField;
                        break;
                }
                break;
        }
    }
    return fields;
}

/// Read CSV file, Excel dialect. Accept "quoted fields ""with quotes"""
std::vector<std::vector<std::string>> readCSV(std::istream& in) {
    std::vector<std::vector<std::string>> table;
    std::string row;
    while (!in.eof()) {
        std::getline(in, row);
        if (in.bad() || in.fail()) {
            break;
        }
        auto fields = readCSVRow(row);
        table.push_back(fields);
    }
    return table;
}

Solution 3: Leveraging Boost Tokenizer with escaped_list_separator

The Boost library offers powerful tools for string manipulation, including its Tokenizer. The escaped_list_separator is specifically designed for CSV-like data.

#include <vector>
#include <string>
#include <boost/tokenizer.hpp>

// Assuming 'line' is a std::string containing a CSV row
std::vector<std::string> parseCSVLine(const std::string& line) {
    std::vector<std::string> vec;
    boost::char_separator<char> sep("," , "", boost::keep_empty_tokens);
    boost::tokenizer<boost::char_separator<char>> tokens(line, sep);
    for (const auto& token : tokens) {
        vec.push_back(token);
    }
    return vec;
}

Solution 5: Advanced Parsing with Boost.Spirit

Boost.Spirit is a powerful C++ library for creating parsers. While it has a steeper learning curve, it offers unparalleled flexibility and performance for complex parsing tasks, including CSV.

#include <boost/spirit/include/qi.hpp>
#include <boost/spirit/include/phoenix_operator.hpp>

#include <iostream>
#include <string>
#include <vector>

namespace qi = boost::spirit::qi;
namespace bascii = boost::spirit::ascii;

template <typename Iterator>
struct csv_parser : qi::grammar<Iterator, std::vector<std::string>(),
    bascii::space_type>
{
    qi::rule<Iterator, char()> COMMA;
    qi::rule<Iterator, char()> DDQUOTE;
    qi::rule<Iterator, std::string(), bascii::space_type> non_escaped;
    qi::rule<Iterator, std::string(), bascii::space_type> escaped;
    qi::rule<Iterator, std::string(), bascii::space_type> field;
    qi::rule<Iterator, std::vector<std::string>(), bascii::space_type> start;

    csv_parser() : csv_parser::base_type(start)
    {
        using namespace qi;
        using qi::lit;
        using qi::lexeme;
        using bascii::char_;

        start = field % ',';
        field = escaped | non_escaped;
        // Handles escaped quotes within quoted fields ("")
        escaped = lexeme['"' >> *(char_(-(char_('"') | ',')) | COMMA | DDQUOTE) >> '"'];
        non_escaped = lexeme[*(char_(-(char_('"') | ',')))];
        DDQUOTE = lit("\"\"")[_val = '"'];
        COMMA = lit(",")[_val = ','];
    }
};

// Example Usage:
int main() {
    std::cout << "Enter CSV lines [empty line to quit]\n";

    using bascii::space;
    typedef std::string::const_iterator iterator_type;
    typedef csv_parser<iterator_type> csv_parser;

    csv_parser grammar;
    std::string str;
    int fid;
    while (std::getline(std::cin, str)) {
        fid = 0;

        if (str.empty())
            break;

        std::vector<std::string> csv;
        std::string::const_iterator it_beg = str.begin();
        std::string::const_iterator it_end = str.end();
        bool r = phrase_parse(it_beg, it_end, grammar, space, csv);

Solution 6: Custom Character-by-Character Parsing (State Machine)

This approach builds a custom parser that processes the CSV data character by character, managing states to correctly interpret fields, quotes, and delimiters. This offers fine-grained control and can be optimized for specific CSV formats.

#include <string>
#include <vector>
#include <istream> // For std::istream

void ParseCSV(const std::string& csvSource, std::vector<std::vector<std::string>>& lines) {
    bool inQuote = false;
    bool newLine = false;
    std::string field;
    lines.clear();
    std::vector<std::string> line;

    std::string::const_iterator aChar = csvSource.begin();
    while (aChar != csvSource.end()) {
        switch (*aChar) {
            case '"':
                newLine = false;
                inQuote = !inQuote;
                break;

            case ',':
                newLine = false;
                if (inQuote) {
                    field += *aChar;
                } else {
                    line.push_back(field);
                    field.clear();
                }
                break;

            case '\n':
            case '\r':
                if (inQuote) {
                    field += *aChar;
                } else {
                    if (!newLine) {
                        line.push_back(field);
                        lines.push_back(line);
                        field.clear();
                        line.clear();
                        newLine = true;
                    }
                }
                break;

            default:
                newLine = false;
                field.push_back(*aChar);
                break;
        }
        aChar++;
    }

    if (field.size())
        line.push_back(field);

    if (line.size())
        lines.push_back(line);
}

Solution 9: C++11 CSV Parser with std::tuple

This approach uses C++11 features like std::tuple and template metaprogramming to create an iterable CSV wrapper, allowing direct mapping of CSV rows to tuples of specified types.

#include <iterator>
#include <sstream>
#include <string>
#include <vector>
#include <tuple> // For std::tuple
#include <iostream> // For std::cout

namespace csvtools {
    // Helper to read the last element of the tuple
    template <std::size_t idx, class... fields>
    typename std::enable_if<idx == std::tuple_size<std::tuple<fields...>>::value - 1>::type
    read_tuple(std::istream& in, std::tuple<fields...>& out, const char delimiter) {
        std::string cell;
        std::getline(in, cell, delimiter);
        std::stringstream cell_stream(cell);
        cell_stream >> std::get<idx>(out);
    }

    // Helper to read subsequent elements of the tuple recursively
    template <std::size_t idx, class... fields>
    typename std::enable_if<idx < std::tuple_size<std::tuple<fields...>>::value - 1>::type
    read_tuple(std::istream& in, std::tuple<fields...>& out, const char delimiter) {
        std::string cell;
        std::getline(in, cell, delimiter);
        std::stringstream cell_stream(cell);
        cell_stream >> std::get<idx>(out);
        read_tuple<idx + 1, fields...>(in, out, delimiter);
    }
}

/// Iterable CSV wrapper around a stream.
template <class... fields>
class csv {
    std::istream& _in;
    const char _delim;
public:
    typedef std::tuple<fields...> value_type;
    class iterator;

    inline csv(std::istream& in, const char delim) : _in(in), _delim(delim) {}

    inline bool good() const { return _in.good(); }
    inline const std::istream& underlying_stream() const { return _in; }

    inline iterator begin();
    inline iterator end();

private:
    inline value_type read_row() {
        std::string line;
        std::getline(_in, line);
        std::stringstream line_stream(line);
        std::tuple<fields...> retval;
        csvtools::read_tuple<0, fields...>(line_stream, retval, _delim);
        return retval;
    }

    /// Iterator class
    class iterator {
        value_type _row;
        csv* _parent;
    public:
        typedef std::input_iterator_tag iterator_category;
        typedef value_type value_type;
        typedef std::size_t difference_type;
        typedef value_type* pointer;
        typedef value_type& reference;

        inline iterator() : _parent(nullptr) {}
        inline iterator(csv& parent) : _parent(parent.good() ? &parent : nullptr) {
            ++(*this); // Read the first row upon construction
        }

        inline iterator& operator++() {
            if (_parent != nullptr) {
                _row = _parent->read_row();
                if (!_parent->good()) {
                    _parent = nullptr; // Reached end of stream
                }
            }
            return *this;
        }

        inline iterator operator++(int) {
            iterator copy = *this;
            ++(*this);
            return copy;
        }

        inline value_type const& operator*() const { return _row; }
        inline value_type const* operator->() const { return &_row; }

        bool operator==(iterator const& other) {
            return (this == &other) || (_parent == nullptr && other._parent == nullptr);
        }
        bool operator!=(iterator const& other) { return !(*this == other); }
    };
};

template <class... fields>
typename csv<fields...>::iterator csv<fields...>::begin() {
    return iterator(*this);
}

template <class... fields>
typename csv<fields...>::iterator csv<fields...>::end() {
    return iterator();
}

Solution 10: Boost.Fusion for Sequence Input/Output

Boost.Fusion can be used to adapt CSV parsing to its sequence I/O mechanisms, allowing for elegant handling of structured data input.

#include <iostream>
#include <sstream>
#include <boost/fusion/adapted/boost_tuple.hpp>
#include <boost/fusion/sequence/io.hpp>
#include <boost/tuple/tuple.hpp> // For boost::tuple

namespace fusion = boost::fusion;

// Custom stream operator to read CSV fields
struct CsvString {
    std::string value;

    friend std::istream& operator>>(std::istream& s, CsvString& v) {
        v.value.clear();
        for (;;) {
            auto c = s.peek();
            if (std::istream::traits_type::eof() == c || ',' == c || '\n' == c)
                break;
            v.value.push_back(c);
            s.get();
        }
        return s;
    }

    friend std::ostream& operator<<(std::ostream& s, CsvString const& v) {
        return s << v.value;
    }
};

Solution 11: File Validation and Basic Parsing

This solution focuses on essential file operations before parsing: checking file existence and validating its extension. The parsing itself involves a simple character replacement to handle commas.

#include <string>
#include <fstream>
#include <iostream>
#include <vector>

bool fileExists(const std::string& fileName) {
    std::ifstream test(fileName.c_str());
    return test.good();
}

bool verifyExtension(const std::string& filename) {
    size_t periodPos = filename.rfind('.');
    if (periodPos == std::string::npos) return false;
    std::string extension = filename.substr(periodPos);
    return extension == ".csv";
}

std::string getExtension(const std::string& filename) {
    size_t periodPos = filename.rfind('.');
    if (periodPos == std::string::npos) return "NO FILE";
    return filename.substr(periodPos);
}

void parseFile(const std::string& fileName) {
    if (fileExists(fileName) && verifyExtension(fileName)) {
        std::ifstream fs(fileName);
        std::string lineCommand;

        while (std::getline(fs, lineCommand)) {
            std::string temp;
            for (char c : lineCommand) {
                if (c != ',') {
                    temp += c;
                } else {
                    temp += " "; // Replace comma with space
                }
            }
            if (!temp.empty() && temp != "\0") {
                // Process the parsed line (temp) here
                std::cout << "Processed line: " << temp << std::endl;
            }
        }
        fs.close();
    } else if (!fileExists(fileName)) {
        std::cerr << "Error: The provided file does not exist: " << fileName << std::endl;
        if (!verifyExtension(fileName)) {
            std::cerr << "\tCheck the file extension." << std::endl;
        }
    } else { // Incorrect extension
        std::cerr << "Incorrect file extension provided for: " << fileName << std::endl;
    }
}

Solution 14: Custom CSV Splitter Function

This solution provides a custom SplitCSV function that handles delimiters and quoted fields, along with a ReadCsv function to manage file reading.

#include <iostream>
#include <vector>
#include <fstream>
#include <string>

std::vector<std::string> SplitCSV(const std::string& data, char separator, char delimiter) {
    std::vector<std::string> values;
    std::string val = "";
    bool inDelimiter = false;
    size_t delimiterCount = 0;
    const char* C = data.c_str();
    size_t P = 0;

    do {
        if ((val.length() == 0) && (C[P] == delimiter)) {
            inDelimiter = !inDelimiter;
            delimiterCount = 0;
            P++;
            continue;
        }
        if (inDelimiter) {
            if (C[P] == delimiter) {
                // Check for escaped delimiter ("")
                if (((delimiterCount % 2) == 0) &&
                    ((C[P + 1] == separator) || (C[P + 1] == 0) || (C[P + 1] == '\n') || (C[P + 1] == '\r'))) {
                    inDelimiter = false;
                    delimiterCount = 0;
                    P++;
                    continue;
                } else {
                    delimiterCount++;
                }
            }
        } else {
            if (C[P] == separator) {
                values.push_back(val);
                val = "";
                P++;
                continue;
            }
            if ((C[P] == 0) || (C[P] == '\n') || (C[P] == '\r'))
                break;
        }
        val += C[P];
        P++;
    } while (P < data.length());
    values.push_back(val);
    return values;
}

bool ReadCsv(const std::string& fname, std::vector<std::vector<std::string>>& data,
             char separator = ',', char delimiter = '"') {
    bool ret = false;
    std::ifstream FCsv(fname);
    if (FCsv) {

        size_t Siz = FCsv.tellg();
        if (Siz > 0) {
            FCsv.seekg(0);

            }
            ret = true;
        }

    }
    return ret;
}

 * @param vls Reference to a vector of strings to hold the parsed line.
 */
void readCSV(FILE* fp, std::vector<std::string>& vls) {
    vls.clear();
    if (!fp) return;
    char buf[10000];
    if (!fgets(buf, sizeof(buf) - 1, fp)) // Read line into buffer
        return;

    std::string s = buf;
    int p, q;
    q = -1;
    // Loop over columns
    while (1) {
        p = q;
        // Find the next comma or newline character
        q = s.find_first_of(",\n", p + 1);
        if (q == std::string::npos)
            break; // End of line reached
        // Extract the substring representing the field
        vls.push_back(s.substr(p + 1, q - p - 1));
    }
}

Solution 17: Custom String Parsing with Whitespace Trimming

This solution focuses on custom parsing logic, handling quoted fields, whitespace trimming, and escaped quotes. It’s designed for flexibility, especially with variations in CSV formatting.

#include <string>

// Trim whitespaces around field or double-quotes, remove double-quotes and replace escaped double-quotes
std::wstring trimquote(const std::wstring& str, const std::wstring& whitespace, const wchar_t quotChar) {
    std::wstring ws;
    std::wstring::size_type strBegin = str.find_first_not_of(whitespace);
    if (strBegin == std::wstring::npos)
        return L"";

    std::wstring::size_type strEnd = str.find_last_not_of(whitespace);
    std::wstring::size_type strRange = strEnd - strBegin + 1;

    if ((str[strBegin] == quotChar) && (str[strEnd] == quotChar)) {
        ws = str.substr(strBegin + 1, strRange - 2);
        strBegin = 0;
        while ((strEnd = ws.find(quotChar, strBegin)) != std::wstring::npos) {
            ws.erase(strEnd, 1); // Remove the quote character
            strBegin = strEnd + 1; // Move past the removed quote
        }
    } else {
        ws = str.substr(strBegin, strRange);
    }
    return ws;
}

// Find the next quoted pair
std::pair<unsigned, unsigned> nextCSVQuotePair(const std::wstring& line, const wchar_t quotChar, unsigned ofs = 0) {
    std::pair<unsigned, unsigned> r;
    r.first = line.find(quotChar, ofs);
    r.second = std::wstring::npos;
    if (r.first != std::wstring::npos) {
        r.second = r.first;
        // Find subsequent quotes, handling escaped quotes ("")
        while (((r.second = line.find(quotChar, r.second + 1)) != std::wstring::npos) &&
               (r.second + 1 < line.length() && line[r.second + 1] == quotChar)) {
            r.second++; // Skip the escaped quote pair
        }
    }
    return r;
}

/**
 * Parses a CSV line into fields.
 * @param fields Output vector to store parsed fields.
 * @param line The input CSV line as a wstring.
 * @return The number of fields parsed.
 */
unsigned parseLine(std::vector<std::wstring>& fields, const std::wstring& line) {
    unsigned ofs, ofs0;
    const wchar_t delim = L',';
    // Define whitespace characters, including Unicode spaces
    const std::wstring whitespace = L" \t\xa0\x3000\x2000\x2001\x2002\x2003\x2004\x2005\x2006\x2007\x2008\x2009\x200a\x202f\x205f";
    const wchar_t quotChar = L'"';
    std::pair<unsigned, unsigned> quot;

    fields.clear();

    ofs = ofs0 = 0;
    quot = nextCSVQuotePair(line, quotChar);
    while ((ofs = line.find(delim, ofs)) != std::wstring::npos) {
        // If the delimiter is inside a quoted field, skip it
        if ((ofs > quot.first) && (ofs < quot.second)) {
            ofs = quot.second + 1; // Move past the closing quote
            quot = nextCSVQuotePair(line, quotChar, ofs); // Find the next quote pair
            continue;
        }
        // Extract and trim the field
        fields.push_back(trimquote(line.substr(ofs0, ofs - ofs0), whitespace, quotChar));
        ofs0 = ofs + 1; // Move past the delimiter
    }
    // Add the last field
    fields.push_back(trimquote(line.substr(ofs0), whitespace, quotChar));

    return fields.size();
}

Boost.Spirit is a powerful C++ library for creating parsers. It can be used to build highly efficient and customizable CSV parsers, capable of handling complex formats according to RFC specifications.

namespace qi = boost::spirit::qi;

template <typename Iterator>
struct csv_parser : qi::grammar<Iterator, std::vector<std::string>(),
    bascii::space_type>
{
    qi::rule<Iterator, char()> COMMA;
    qi::rule<Iterator, char()> DDQUOTE;
    qi::rule<Iterator, std::string(), bascii::space_type> non_escaped;
    qi::rule<Iterator, std::string(), bascii::space_type> escaped;
    qi::rule<Iterator, std::string(), bascii::space_type> field;
    qi::rule<Iterator, std::vector<std::string>(), bascii::space_type> start;

    csv_parser() : csv_parser::base_type(start)
    {
        using namespace qi;
        using qi::lit;
        using qi::lexeme;
        using bascii::char_;

        start = field % ',';
        field = escaped | non_escaped;
        // Handles escaped quotes ("") within quoted fields
        escaped = lexeme['"' >> *(char_(-(char_('"') | ',')) | COMMA | DDQUOTE) >> '"'];
        non_escaped = lexeme[*(char_(-(char_('"') | ',')))];
        DDQUOTE = lit("\"\"")[_val = '"']; // Represents an escaped quote
        COMMA = lit(",")[_val = ','];     // Represents a comma within a quoted field
    }
};

Solution 20: Character-by-Character Parsing for Large Files

This method reads the file character by character, processing one row at a time. This approach is suitable for very large files as it avoids loading the entire file into memory.

#include <vector>
#include <string>
#include <fstream>
#include <iostream>

class read_csv_t {
public:
    read_csv_t() {} // Default constructor

    // Opens the CSV file. Returns 0 on success, -1 on failure.
    int open(const std::string& file_name) {
        m_ifs.open(file_name);
        if (!m_ifs.is_open()) {
            return -1; // Indicate failure
        }
        return 0; // Indicate success
    }

    // Reads a single row from the CSV file.
    // Returns a vector of strings representing the columns in the row.
    // Returns an empty vector if end-of-file is reached or an error occurs.
    std::vector<std::string> read_row() {
        bool quote_mode = false;
        std::vector<std::string> row;
        std::string column;
        char c;

        while (m_ifs.get(c)) {
            switch (c) {
                case ',':
                    if (quote_mode) {
                        column += c; // Add comma to column if inside quotes
                    } else {
                        row.push_back(column); // End of column, add to row
                        column.clear();
                    }
                    break;
                case '"':
                    quote_mode = !quote_mode; // Toggle quote mode
                    break;
                case '\n':
                case '\r':
                    if (quote_mode) {
                        column += c; // Add newline to column if inside quotes
                    } else {
                        // End of row detected outside quotes
                        return row; // Return the completed row
                    }
                    break;
                default:
                    column += c; // Add character to the current column
                    break;
            }
        }

        // End of file reached. Add the last column if it exists.
        if (!column.empty() || !row.empty()) {
             row.push_back(column);
        }
        
        // Close the file stream after reading all rows.
        m_ifs.close(); 
        return row; // Return the final row or an empty vector if EOF was clean
    }

private:
    std::ifstream m_ifs;
};
