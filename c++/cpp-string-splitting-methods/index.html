“C++ String Splitting Techniques: Methods and Examples”

Splitting a string into smaller parts based on a delimiter is a common programming task. C++ offers several ways to achieve this, ranging from manual iteration to utilizing powerful libraries. Below are various methods with practical code examples.

Solution 1: Using Boost.Tokenizer

The Boost library provides a robust and flexible tokenizer class that simplifies string splitting.

  • C++11 Version:

    #include <iostream>
#include <string>
#include <boost/tokenizer.hpp>

int main() {
    std::string text = "token, test   string";
    boost::char_separator<char> sep(", ");
    boost::tokenizer<boost::char_separator<char>> tokens(text, sep);

    for (const auto& t : tokens) {
        std::cout << t << "." << std::endl;
    }
    return 0;
}

  • Pre-C++11 Version (using BOOST_FOREACH):

    #include <iostream>
#include <string>
#include <boost/foreach.hpp>
#include <boost/tokenizer.hpp>

int main() {
    std::string text = "token, test   string";
    boost::char_separator<char> sep(", ");
    boost::tokenizer<boost::char_separator<char>> tokens(text, sep);

    BOOST_FOREACH (const std::string& t, tokens) {
        std::cout << t << "." << std::endl;
    }
    return 0;
}

Solution 2: Manual Splitting with std::vector

A straightforward approach involves manually iterating through the string and extracting substrings.

#include <vector>
#include <string>

std::vector<std::string> split(const char *str, char c = ' ') {
    std::vector<std::string> result;
    do {
        const char *begin = str;
        while(*str != c && *str)
            str++;
        result.push_back(std::string(begin, str));
    } while (0 != *str++);
    return result;
}

Solution 5: Using strtok (C-style)

The C standard library function strtok can be used, but it modifies the original string and has thread-safety considerations.

#include <cstdio> // For printf
#include <cstring> // For strtok

int main() {
    char myString[] = "The quick brown fox";
    char *p = strtok(myString, " ");
    while (p) {
        printf("Token: %s\n", p);
        p = strtok(NULL, " ");
    }
    return 0;
}

Caveats: strtok modifies the string by inserting null terminators. It’s not thread-safe and requires a mutable string.

Solution 6: Using std::regex_token_iterator

For more complex splitting criteria, especially those involving patterns, std::regex_token_iterator is highly effective.

#include <iostream>

Solution 7: Custom split Function using find and substr

A common pattern involves repeatedly finding the delimiter and extracting substrings.

#include <vector>
#include <string>

void split(std::vector<std::string>& theStringVector,
           const std::string& theString,
           const std::string& theDelimiter) {
    size_t start = 0, end = 0;

    while (end != std::string::npos) {
        end = theString.find(theDelimiter, start);
        theStringVector.push_back(theString.substr(start, (end == std::string::npos) ? std::string::npos : end - start));
        start = ((end > (std::string::npos - theDelimiter.size())) ? std::string::npos : end + theDelimiter.size());
    }
}

int main() {
    std::vector<std::string> v;
    split(v, "A:PEP:909:Inventory Item", ":");
    for (const auto& s : v) {
        std::cout << s << std::endl;
    }
    return 0;
}

Solution 9: STL-only approach using find_first_not_of and find

This method handles consecutive delimiters and leading/trailing delimiters effectively.

#include <string>
#include <vector>
#include <iostream>

const char DELIMITER = ' '; // Example delimiter

void tokenize(std::string str, std::vector<std::string>& token_v) {
    size_t start = str.find_first_not_of(DELIMITER), end = start;

    while (start != std::string::npos) {
        end = str.find(DELIMITER, start);
        token_v.push_back(str.substr(start, end - start));
        start = str.find_first_not_of(DELIMITER, end);
    }
}

int main() {
    std::string my_string = "This is a test string";
    std::vector<std::string> tokens;
    tokenize(my_string, tokens);
    for(const auto& t : tokens) {
        std::cout << t << std::endl;
    }
    return 0;
}

Solution 11: Custom Tokenizer Class

For more structured control over the tokenization process, a dedicated class can be beneficial.

// Header file
class Tokenizer {
public:
    static const std::string DELIMITERS;
    Tokenizer(const std::string& str);
    Tokenizer(const std::string& str, const std::string& delimiters);
    bool NextToken();
    bool NextToken(const std::string& delimiters);
    const std::string GetToken() const;
    void Reset();
protected:
    size_t m_offset;
    const std::string m_string;
    std::string m_token;
    std::string m_delimiters;
};

// CPP file
const std::string Tokenizer::DELIMITERS(" \t\n\r");

Tokenizer::Tokenizer(const std::string& s) : m_string(s), m_offset(0), m_delimiters(DELIMITERS) {}
Tokenizer::Tokenizer(const std::string& s, const std::string& delimiters) : m_string(s), m_offset(0), m_delimiters(delimiters) {}

bool Tokenizer::NextToken() { return NextToken(m_delimiters); }

bool Tokenizer::NextToken(const std::string& delimiters) {
    size_t i = m_string.find_first_not_of(delimiters, m_offset);
    if (std::string::npos == i) {
        m_offset = m_string.length();
        return false;
    }
    size_t j = m_string.find_first_of(delimiters, i);
    if (std::string::npos == j) {
        m_token = m_string.substr(i);
        m_offset = m_string.length();
        return true;
    }
    m_token = m_string.substr(i, j - i);
    m_offset = j;
    return true;
}

// Example Usage:
int main() {
    std::vector<std::string> v;
    Tokenizer s("split this string", " ");
    while (s.NextToken()) {
        v.push_back(s.GetToken());
    }
    // Process vector v
    return 0;
}

Solution 15: String Manipulation with Iterators and find

This approach leverages standard library iterators for string traversal.

#include <string>
#include <vector>
#include <iostream>
#include <algorithm> // For std::find

int main() {
    std::string str = "The quick brown fox";
    std::vector<std::string> tokens;

    auto start = std::cbegin(str);
    auto end = std::cend(str);

    while (start != end) {
        auto delim_pos = std::find(start, end, ' ');
        tokens.emplace_back(start, delim_pos);
        start = delim_pos;
        if (start != end) {
            start++; // Move past the delimiter
        }
    }

    for (const auto& token : tokens) {
        std::cout << token << std::endl;
    }
    return 0;
}

Solution 16: strtok with Null Handling for C++

A thread-safe and C++ idiomatic way to use strtok.

std::vector<std::string> split_strtok(char* str, const char* delim) {
    std::vector<std::string> tokens;
    char* token = std::strtok(str, delim);
    while (token != nullptr) {
        tokens.push_back(token);
        token = std::strtok(nullptr, delim);
    }
    return tokens;
}

Solution 20: Custom Tokenizer with Delimiter Handling

A flexible function that allows control over discarding empty tokens.

#include <string>
#include <vector>
#include <iostream>

unsigned TokenizeString(const std::string& i_source,
                        const std::string& i_seperators,
                        bool i_discard_empty_tokens,
                        std::vector<std::string>& o_tokens) {
    unsigned prev_pos = 0, pos = 0, number_of_tokens = 0;
    o_tokens.clear();
    pos = i_source.find_first_of(i_seperators, pos);

    while (pos != std::string::npos) {
        std::string token = i_source.substr(prev_pos, pos - prev_pos);
        if (!i_discard_empty_tokens || token != "") {
            o_tokens.push_back(token);
            number_of_tokens++;
        }
        pos++;
        prev_pos = pos;
        pos = i_source.find_first_of(i_seperators, pos);
    }

    if (prev_pos < i_source.length()) {
        o_tokens.push_back(i_source.substr(prev_pos));
        number_of_tokens++;
    }
    return number_of_tokens;
}

int main() {
    std::string source = "One,,Two, Three";
    std::vector<std::string> tokens;
    TokenizeString(source, ",", true, tokens); // Discard empty tokens
    for (const auto& t : tokens) {
        std::cout << t << std::endl;
    }
    return 0;
}

Solution 23: Generic Separator Splitting Function

A flexible function to split by any given separator.

#include <vector>
#include <string>
#include <iostream>

template <typename valueType>
static std::vector<valueType> Split(valueType text, const valueType& delimiter) {
    std::vector<valueType> tokens;
    size_t pos = 0;
    valueType token;

    while ((pos = text.find(delimiter)) != valueType::npos) {
        token = text.substr(0, pos);
        tokens.push_back(token);
        text.erase(0, pos + delimiter.length());
    }
    tokens.push_back(text); // Add the last part
    return tokens;
}

Solution 24: Advanced Regex Tokenization with Quote Handling

This approach handles quoted strings and various delimiters using regular expressions.

#include <string>
#include <locale>
#include <regex>
#include <vector>
#include <iostream>

std::vector<std::wstring> tokenize_string(std::wstring string_to_tokenize) {
    std::vector<std::wstring> tokens;
    std::wregex re(LR"("[^"]*"|'[^']*'|[^"' ]+)", std::regex_constants::collate);
    std::wsregex_iterator next(string_to_tokenize.begin(), string_to_tokenize.end(), re, std::regex_constants::match_not_null);
    std::wsregex_iterator end;
    const wchar_t single_quote = L'\'';
    const wchar_t double_quote = L'\"';

    while (next != end) {
        std::wsmatch match = *next;
        const std::wstring token = match.str(0);
        next++;

        if (token.length() > 2 && (token.front() == double_quote || token.front() == single_quote)) {
            tokens.emplace_back(token.begin() + 1, token.begin() + token.length() - 1);
        } else {
            tokens.emplace_back(token);
        }
    }
    return tokens;
}

int main() {
    std::wstring input = L"split \"this string\" 'with quotes'";
    std::vector<std::wstring> result = tokenize_string(input);
    for (const auto& t : result) {
        std::wcout << t << std::endl;
    }
    return 0;
}

Solution 25: boost::make_find_iterator

Leveraging Boost’s find_iterator for efficient delimiter searching.

#include <vector>
#include <string>
#include <boost/algorithm/string.hpp>
#include <boost/iterator/find_iterator.hpp>
#include <iostream>

template<typename CH>
std::vector<std::basic_string<CH>> tokenize(
    const std::basic_string<CH>& Input,
    const std::basic_string<CH>& Delimiter,
    bool remove_empty_token) {

    typedef typename std::basic_string<CH>::const_iterator string_iterator_t;
    typedef boost::find_iterator<string_iterator_t> string_find_iterator_t;

    std::vector<std::basic_string<CH>> Result;
    string_iterator_t it = Input.begin();
    string_iterator_t it_end = Input.end();

    for (string_find_iterator_t i = boost::make_find_iterator(Input, boost::first_finder(Delimiter, boost::is_equal()));
         i != string_find_iterator_t();
         ++i) {
        if (remove_empty_token) {
            if (it != i->begin())
                Result.push_back(std::basic_string<CH>(it, i->begin()));
        } else {
            Result.push_back(std::basic_string<CH>(it, i->begin()));
        }
        it = i->end();
    }
    if (it != it_end) {
        Result.push_back(std::basic_string<CH>(it, it_end));
    }
    return Result;
}

int main() {
    std::string s = "a,b,,c";
    std::vector<std::string> tokens = tokenize(s, ",", false); // Keep empty tokens
    for (const auto& t : tokens) {
        std::cout << "\"" << t << "\"" << std::endl;
    }
    return 0;
}

Solution 27: Simple StrTokenizer Function

A straightforward function for tokenizing strings.

#include <string>
#include <vector>
#include <iostream>

void StrTokenizer(std::string& source, const char* delimiter, std::vector<std::string>& Tokens) {
    size_t new_index = 0, old_index = 0;
    while (new_index != std::string::npos) {
        new_index = source.find(delimiter, old_index);
        Tokens.emplace_back(source.substr(old_index, new_index - old_index));
        if (new_index != std::string::npos) {
            old_index = ++new_index;
        }
    }
}

int main() {
    std::string source = "One delimiter test string";
    std::vector<std::string> tokens;
    StrTokenizer(source, " ", tokens);
    for (const auto& t : tokens) {
        std::cout << t << std::endl;
    }
    return 0;
